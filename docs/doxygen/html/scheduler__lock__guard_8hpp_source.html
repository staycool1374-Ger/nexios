63class SchedulerLockGuard {
64 public:
66 explicit SchedulerLockGuard(sync::SpinLock &lock) noexcept
67 : lock_(lock), owns_(true) {
68 lock_.lock();
69 ++scheduler_lock_depth[arch::cpu_index()];
70 }
71
73 explicit SchedulerLockGuard(sync::SpinLock &lock,
74 adopt_lock_t) noexcept
75 : lock_(lock), owns_(true) {
76 // Adopt: the caller holds the lock (a racing take by another
77 // CPU blocks or fails its try; only this CPU's paths index
78 // this CPU's slot), so marking here is race-free.
79 ++scheduler_lock_depth[arch::cpu_index()];
80 }
81 ~SchedulerLockGuard() noexcept {
82 if (owns_) {
83 // Unmark BEFORE release: with the slot clear first, a racing
84 // same-CPU observer sees depth 0 only when the lock is
85 // actually free-or-freeing (a try_lock in between still
86 // fails — the lock itself is still held here).
87 --scheduler_lock_depth[arch::cpu_index()];
88 lock_.unlock();
89 }
90 }
91
92 bool owns_lock() const noexcept { return owns_; }
94 void unlock() noexcept {
95 if (owns_) {
96 owns_ = false;
97 --scheduler_lock_depth[arch::cpu_index()];
98 lock_.unlock();
99 }
100 }
101 SchedulerLockGuard(const SchedulerLockGuard &) = delete;
102 SchedulerLockGuard &operator=(const SchedulerLockGuard &) = delete;
103 SchedulerLockGuard(SchedulerLockGuard &&) = delete;
104 SchedulerLockGuard &operator=(SchedulerLockGuard &&) = delete;
105
106 private:
107 sync::SpinLock &lock_;
108 bool owns_ = false;
109};
