344 template <size_t Capacity> class DmesgBuffer {
345 static_assert((Capacity & (Capacity - 1)) == 0,
346 "Capacity must be power of 2");
347 static constexpr size_t MASK = Capacity - 1;
348
349 LogEntry buffer[Capacity];
350 alignas(64) volatile size_t head{0};
351 alignas(64) volatile size_t tail{0};
352
353 public:
356 bool push(ErrorSubsystem subsys, uint64_t err_code, LogSeverity sev,
357 const char *msg, uintptr_t ctx = 0) {
358 if (s_suppressed_)
359 return true;
360 const uint64_t ts = arch::Timer::ticks();
361 const uint64_t tid = kernel::Scheduler::current_task()
362 ? kernel::Scheduler::current_task()->id
363 : 0;
364
365 size_t h = atomic_load(&head, __ATOMIC_RELAXED);
366 size_t t = atomic_load(&tail, __ATOMIC_ACQUIRE);
367
368 size_t next_h = (h + 1) & MASK;
369 bool overwrote = false;
370
371 if (next_h == t) {
372 overwrote = true;
373 atomic_store(&tail, (t + 1) & MASK, __ATOMIC_RELEASE);
374 }
375
376 buffer[h] = LogEntry{};
377 buffer[h].timestamp = ts;
378 buffer[h].wall_ms = current_wall_ms();
379 buffer[h].task_id = tid;
380 buffer[h].subsystem = subsys;
381 buffer[h].severity = sev;
382 buffer[h].error_code = err_code;
383 buffer[h].context = ctx;
384 // Owned copy (SIL3): the entry stores a bounded char array, so
385 // producers may pass transient/ring buffers (e.g. the ELF loader's
386 // message slots) without dangling once the source is reused.
387 size_t i = 0;
388 if (msg) {
389 while (msg[i] && i < LogEntry::kMessageCap - 1) {
390 buffer[h].message[i] = msg[i];
391 ++i;
392 }
393 }
394 buffer[h].message[i] = '\0';
395
396 atomic_store(&head, next_h, __ATOMIC_RELEASE);
397 return !overwrote;
398 }
399
402 bool pop(LogEntry &entry) {
403 size_t t = atomic_load(&tail, __ATOMIC_RELAXED);
404 size_t h = atomic_load(&head, __ATOMIC_ACQUIRE);
405
406 if (t == h)
407 return false;
408
409 entry = buffer[t];
410 atomic_store(&tail, (t + 1) & MASK, __ATOMIC_RELEASE);
411 return true;
412 }
413
415 template <typename Fn> void for_each(Fn &&fn) const {
416 size_t t = atomic_load(&tail, __ATOMIC_ACQUIRE);
417 size_t h = atomic_load(&head, __ATOMIC_ACQUIRE);
418
419 while (t != h) {
420 fn(buffer[t]);
421 t = (t + 1) & MASK;
422 }
423 }
424
426 bool empty() const {
427 return atomic_load(&head, __ATOMIC_ACQUIRE) ==
428 atomic_load(&tail, __ATOMIC_ACQUIRE);
429 }
430
432 size_t size() const {
433 size_t h = atomic_load(&head, __ATOMIC_ACQUIRE);
434 size_t t = atomic_load(&tail, __ATOMIC_ACQUIRE);
435 return (h - t) & MASK;
436 }
437
439 void clear() {
440 size_t h = atomic_load(&head, __ATOMIC_RELAXED);
441 atomic_store(&tail, h, __ATOMIC_RELEASE);
442 }
443
445 static void set_suppressed(bool v) {
446 s_suppressed_ = v;
447 }
449 static bool is_suppressed() {
450 return s_suppressed_;
451 }
452
453 size_t head_index() const {
454 return head;
455 }
456 size_t tail_index() const {
457 return tail;
458 }

464 false
465#else
466 true

496inline bool base_code_is_info(uint64_t code) {
497 if ((code & ~0xFFULL) == 0xDA00 || (code & ~0xFFULL) == 0xDB00 ||
498 (code & ~0xFFULL) == 0xDC00) {
499 return true;
500 }
501 switch (code) {
502 case kDmesgBase_DAEMON + 2:
503 case kDmesgBase_DAEMON + 3:
504 case kDmesgBase_DAEMON + 4:
505 case kDmesgBase_DAEMON + 5:
506 case kDmesgBase_DAEMON + 6:
507 case kDmesgBase_ELF + 1:
508 case kDmesgBase_ELF + 2:
509 case kDmesgBase_ELF + 3:
510 case kDmesgBase_USER + 1:
511 return true;
512 default:
513 return false;
514 }

528inline const char *base_error_string(uint64_t code) {
529 // Panic record (pushed by panic() as BASE/FATAL, issue #234).
530 if (code == kDmesgPanicCode) {
531 return "Kernel panic";
532 }
533 // Custom event ranges
534 if ((code & ~0xFFULL) == 0xDA00) {
535 switch (code) {
536 case 0xDA01:
537 return "Daemon exited";
538 case 0xDA02:
539 return "Daemon restarted";
540 case 0xDA03:
541 return "Daemon ensured";
542 case 0xDA04:
543 return "Daemon terminated";
544 case 0xDA05:
545 return "Daemon restarting";
546 case 0xDA06:
547 return "Daemon up";
548 default:
549 break;
550 }
551 return "Daemon event";
552 }
553 if ((code & ~0xFFULL) == 0xDB00) {
554 switch (code) {
555 case 0xDB01:
556 return "ELF load started";
557 case 0xDB02:
558 return "ELF load completed";
559 case 0xDB03:
560 return "ELF load canceled";
561 case 0xDB04:
562 return "ELF load failed: invalid elf-file";
563 case 0xDB05:
564 return "ELF load failed: not enough memory";
565 case 0xDB06:
566 return "ELF load failed: file not found";
567 case 0xDB07:
568 return "ELF load failed: read error";
569 case 0xDB08:
570 return "ELF load rejected: already loading";
571 case 0xDB09:
572 return "ELF load rejected: not loading";
573 case 0xDB0C:
574 return "ELF authenticity mismatch: task terminated";
575 case 0xDB0D:
576 return "ELF load failed: authenticity baseline";
577 default:
578 break;
579 }
580 return "ELF loader event";
581 }
582 if ((code & ~0xFFULL) == 0xDC00) {
583 switch (code) {
584 case 0xDC01:
585 return "Task exited";
586 case 0xDC02:
587 return "Task faulted";
588 default:
589 break;
590 }
591 return "Task end event";
592 }
593 // Standard kernel::Error range (0–9)
594 switch (static_cast<kernel::Error>(code)) {
595 case kernel::Error::OK:
596 return "OK";
597 case kernel::Error::OOM:
598 return "Out of memory";
599 case kernel::Error::INVALID_ARG:
600 return "Invalid argument";
601 case kernel::Error::NOT_FOUND:
602 return "Not found";
603 case kernel::Error::ALREADY_EXISTS:
604 return "Already exists";
605 case kernel::Error::TIMEOUT:
606 return "Timeout";
607 case kernel::Error::BUSY:
608 return "Busy";
609 case kernel::Error::NOT_IMPLEMENTED:
610 return "Not implemented";
611 case kernel::Error::IO_ERROR:
612 return "I/O error";
613 case kernel::Error::CORRUPTED:
614 return "Corrupted";
615 }
616 return "Unknown base error";
617}

668inline uint64_t strip_canonical(ErrorSubsystem subsys, uint64_t code) {
669 if (subsys == ErrorSubsystem::BASE) {
670 return code;

672 const uint64_t base = subsystem_base(subsys);
673 if (code >= base && code < base + kDmesgStride) {

684 if (catalog::catalog_lookup(subsys, code) != nullptr) {
