175 template <size_t Capacity> class DmesgBuffer {
176 static_assert((Capacity & (Capacity - 1)) == 0,
177 "Capacity must be power of 2");
178 static constexpr size_t MASK = Capacity - 1;
179
180 LogEntry buffer[Capacity];
181 alignas(64) volatile size_t head{0};
182 alignas(64) volatile size_t tail{0};
183
184 public:
187 bool push(ErrorSubsystem subsys, uint64_t err_code, const char *msg,
188 uintptr_t ctx = 0) {
189 if (s_suppressed_)
190 return true;
191 const uint64_t ts = arch::Timer::ticks();
192 const uint64_t tid = kernel::Scheduler::current_task()
193 ? kernel::Scheduler::current_task()->id
194 : 0;
195
196 size_t h = atomic_load(&head, __ATOMIC_RELAXED);
197 size_t t = atomic_load(&tail, __ATOMIC_ACQUIRE);
198
199 size_t next_h = (h + 1) & MASK;
200 bool overwrote = false;
201
202 if (next_h == t) {
203 overwrote = true;
204 atomic_store(&tail, (t + 1) & MASK, __ATOMIC_RELEASE);
205 }
206
207 buffer[h] = LogEntry{};
208 buffer[h].timestamp = ts;
209 buffer[h].task_id = tid;
210 buffer[h].subsystem = subsys;
211 buffer[h].error_code = err_code;
212 buffer[h].context = ctx;
213 // Owned copy (SIL3): the entry stores a bounded char array, so
214 // producers may pass transient/ring buffers (e.g. the ELF loader's
215 // message slots) without dangling once the source is reused.
216 size_t i = 0;
217 if (msg) {
218 while (msg[i] && i < LogEntry::kMessageCap - 1) {
219 buffer[h].message[i] = msg[i];
220 ++i;
221 }
222 }
223 buffer[h].message[i] = '\0';
224
225 atomic_store(&head, next_h, __ATOMIC_RELEASE);
226 return !overwrote;
227 }
228
231 bool pop(LogEntry &entry) {
232 size_t t = atomic_load(&tail, __ATOMIC_RELAXED);
233 size_t h = atomic_load(&head, __ATOMIC_ACQUIRE);
234
235 if (t == h)
236 return false;
237
238 entry = buffer[t];
239 atomic_store(&tail, (t + 1) & MASK, __ATOMIC_RELEASE);
240 return true;
241 }
242
244 template <typename Fn> void for_each(Fn &&fn) const {
245 size_t t = atomic_load(&tail, __ATOMIC_ACQUIRE);
246 size_t h = atomic_load(&head, __ATOMIC_ACQUIRE);
247
248 while (t != h) {
249 fn(buffer[t]);
250 t = (t + 1) & MASK;
251 }
252 }
253
255 bool empty() const {
256 return atomic_load(&head, __ATOMIC_ACQUIRE) ==
257 atomic_load(&tail, __ATOMIC_ACQUIRE);
258 }
259
261 size_t size() const {
262 size_t h = atomic_load(&head, __ATOMIC_ACQUIRE);
263 size_t t = atomic_load(&tail, __ATOMIC_ACQUIRE);
264 return (h - t) & MASK;
265 }
266
268 void clear() {
269 size_t h = atomic_load(&head, __ATOMIC_RELAXED);
270 atomic_store(&tail, h, __ATOMIC_RELEASE);
271 }
272
274 static void set_suppressed(bool v) {
275 s_suppressed_ = v;
276 }
278 static bool is_suppressed() {
279 return s_suppressed_;
280 }
281
282 size_t head_index() const {
283 return head;
284 }
285 size_t tail_index() const {
286 return tail;
287 }

321inline const char *base_error_string(uint64_t code) {
322 // Custom event ranges
323 if ((code & ~0xFFULL) == 0xDA00) {
324 switch (code) {
325 case 0xDA01:
326 return "Daemon exited";
327 case 0xDA02:
328 return "Daemon restarted";
329 case 0xDA03:
330 return "Daemon ensured";
331 case 0xDA04:
332 return "Daemon terminated";
333 case 0xDA05:
334 return "Daemon restarting";
335 case 0xDA06:
336 return "Daemon up";
337 default:
338 break;
339 }
340 return "Daemon event";
341 }
342 if ((code & ~0xFFULL) == 0xDB00) {
343 switch (code) {
344 case 0xDB01:
345 return "ELF load started";
346 case 0xDB02:
347 return "ELF load completed";
348 case 0xDB03:
349 return "ELF load canceled";
350 case 0xDB04:
351 return "ELF load failed: invalid elf-file";
352 case 0xDB05:
353 return "ELF load failed: not enough memory";
354 case 0xDB06:
355 return "ELF load failed: file not found";
356 case 0xDB07:
357 return "ELF load failed: read error";
358 case 0xDB08:
359 return "ELF load rejected: already loading";
360 case 0xDB09:
361 return "ELF load rejected: not loading";
362 default:
363 break;
364 }
365 return "ELF loader event";
366 }
367 // Standard kernel::Error range (0–9)
368 switch (static_cast<kernel::Error>(code)) {
369 case kernel::Error::OK:
370 return "OK";
371 case kernel::Error::OOM:
372 return "Out of memory";
373 case kernel::Error::INVALID_ARG:
374 return "Invalid argument";
375 case kernel::Error::NOT_FOUND:
376 return "Not found";
377 case kernel::Error::ALREADY_EXISTS:
378 return "Already exists";
379 case kernel::Error::TIMEOUT:
380 return "Timeout";
381 case kernel::Error::BUSY:
382 return "Busy";
383 case kernel::Error::NOT_IMPLEMENTED:
384 return "Not implemented";
385 case kernel::Error::IO_ERROR:
386 return "I/O error";
387 case kernel::Error::CORRUPTED:
388 return "Corrupted";
389 }
390 return "Unknown base error";
391}
