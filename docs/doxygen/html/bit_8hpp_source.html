40template <typename To, TriviallyCopiable From>
41[[nodiscard]] constexpr To bit_cast(const From &src) noexcept {
42 static_assert(sizeof(To) == sizeof(From),
43 "bit_cast requires equal-sized types");
44 static_assert(__is_trivially_copyable(To),
45 "bit_cast destination must be trivially copyable");
46 // __builtin_memcpy is constant-evaluated; alignment-safe by construction
47 // (bytewise copy, no actual misaligned access).
48 To dst{};
49 __builtin_memcpy(&dst, &src, sizeof(To));
50 return dst;
51}

65[[nodiscard]] constexpr uint16_t byteswap16(uint16_t value) noexcept {
66 return static_cast<uint16_t>((value << 8) | (value >> 8));
67}

70[[nodiscard]] constexpr uint32_t byteswap32(uint32_t value) noexcept {
71 return ((value & 0x000000FFULL) << 24) | ((value & 0x0000FF00ULL) << 8) |
72 ((value & 0x00FF0000ULL) >> 8) | ((value & 0xFF000000ULL) >> 24);
73}
74
76[[nodiscard]] constexpr uint64_t byteswap64(uint64_t value) noexcept {
77 return ((value & 0x00000000000000FFULL) << 56) |
78 ((value & 0x000000000000FF00ULL) << 40) |
79 ((value & 0x0000000000FF0000ULL) << 24) |
80 ((value & 0x00000000FF000000ULL) << 8) |
81 ((value & 0x000000FF00000000ULL) >> 8) |
82 ((value & 0x0000FF0000000000ULL) >> 24) |
83 ((value & 0x00FF000000000000ULL) >> 40) |
84 ((value & 0xFF00000000000000ULL) >> 56);
85}
