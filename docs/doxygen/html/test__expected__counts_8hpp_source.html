343inline size_t expected_for_class(const char *name) {
344 for (size_t i = 0; i < k_expected_count_size; ++i) {
345 if (__builtin_strcmp(name, k_expected_counts[i].name) == 0) {
346 return arch_count(k_expected_counts[i]);
347 }
348 }
349 return 0;
350}

352inline bool validate_class_count(const char *name, size_t actual_count) {
353 size_t expected = expected_for_class(name);
354 if (expected == 0) {
355 return true;
356 }
357 if (actual_count != expected) {
358 Logger::warn("[TCOUNT] MISMATCH class=%s expected=%u actual=%u "
359 " -- update test_expected_counts.hpp",
360 name, (unsigned)expected, (unsigned)actual_count);
361 return false;
362 }
363 return true;
364}

366inline void validate_all_consistency() {
367 // Issue #173: no `all` class anymore. Every fine-grained class must
368 // be covered by at least one aggregate or by a scripted standalone
369 // special; `safe`/`selftest` are curated overlapping subsets and the
370 // foreign-arch classes are covered off-arch, so all four groups are
371 // excluded from the fine side of the comparison. Intentional overlap
372 // is allowed and reported: hal_apic + apic_tpr are members of both
373 // `hal` and `smp` (HAL devices AND SMP coordination primitives), and
374 // `testrunner` names both a fine class and its aggregate.
375 static constexpr const char *k_aggregates[] = {
376 "core_basic", "core_sync", "core_sched", "core_task",
377 "core_memory", "core_syscall", "core_tracing",
378 "ipc", "capability", "proc_elf",
379 "storage", "servers", "drivers", "hal",
380 "smp", "smp_multicpu", "deadline", "ui",
381 "logging_debug", "random", "bench", "testrunner",
382 nullptr,
383 };
384 static constexpr const char *k_specials[] = {
385 "ahci_live", "iommu_live", "task_fpu", "task_tcb_log",
386 nullptr,
387 };
388 size_t sum_covered = 0;
389 size_t sum_fine = 0;
390 size_t sum_singletons = 0; // testrunner fine + specials: counted
391 // once each on the fine side
392 for (size_t i = 0; i < k_expected_count_size; ++i) {
393 const char *name = k_expected_counts[i].name;
394 size_t c = arch_count(k_expected_counts[i]);
395 if (__builtin_strcmp(name, "safe") == 0 ||
396 __builtin_strcmp(name, "selftest") == 0 ||
397 __builtin_strcmp(name, "arch_aarch64") == 0 ||
398 __builtin_strcmp(name, "arch_riscv64") == 0) {
399 continue;
400 }
401 bool is_covered = false;
402 bool is_singleton = false;
403 for (const char *const *p = k_aggregates; *p != nullptr; ++p) {
404 if (__builtin_strcmp(name, *p) == 0) {
405 is_covered = true;
406 break;
407 }
408 }
409 if (!is_covered) {
410 for (const char *const *p = k_specials; *p != nullptr; ++p) {
411 if (__builtin_strcmp(name, *p) == 0) {
412 is_covered = true;
413 is_singleton = true;
414 break;
415 }
416 }
417 } else if (__builtin_strcmp(name, "testrunner") == 0) {
418 is_singleton = true;
419 }
420 if (is_covered) {
421 sum_covered += c;
422 } else {
423 sum_fine += c;
424 }
425 if (is_singleton) {
426 sum_singletons += c;
427 }
428 }
429 if (sum_covered < sum_fine + sum_singletons) {
430 Logger::warn("[TCOUNT] CONSISTENCY: aggregates+specials=%u < "
431 "fine+singletons=%u -- a fine class is uncovered",
432 (unsigned)sum_covered,
433 (unsigned)(sum_fine + sum_singletons));
434 } else {
435 Logger::info(
436 "[TCOUNT] CONSISTENCY: aggregates+specials=%u >= "
