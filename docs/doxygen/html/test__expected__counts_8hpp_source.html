246inline size_t expected_for_class(const char *name) {
247 for (size_t i = 0; i < k_expected_count_size; ++i) {
248 if (__builtin_strcmp(name, k_expected_counts[i].name) == 0) {
249 return arch_count(k_expected_counts[i]);
250 }
251 }
252 return 0;
253}

255inline bool validate_class_count(const char *name, size_t actual_count) {
256 size_t expected = expected_for_class(name);
257 if (expected == 0) {
258 return true;
259 }
260 if (actual_count != expected) {
261 Logger::warn("[TCOUNT] MISMATCH class=%s expected=%u actual=%u "
262 " -- update test_expected_counts.hpp",
263 name, (unsigned)expected, (unsigned)actual_count);
264 return false;
265 }
266 return true;
267}

269inline void validate_all_consistency() {
270 size_t all_count = 0;
271 size_t sum_individual = 0;
272 for (size_t i = 0; i < k_expected_count_size; ++i) {
273 size_t c = arch_count(k_expected_counts[i]);
274 if (__builtin_strcmp(k_expected_counts[i].name, "all") == 0) {
275 all_count = c;
276 } else if (__builtin_strcmp(k_expected_counts[i].name, "safe") != 0) {
277 sum_individual += c;
278 }
279 }
280 if (all_count > 0 && sum_individual < all_count) {
281 Logger::warn("[TCOUNT] CONSISTENCY: sum(individual)=%u < all=%u -- "
282 "some tests missing from individual class entries",
283 (unsigned)sum_individual, (unsigned)all_count);
284 } else if (all_count > 0) {
285 Logger::info(
286 "[TCOUNT] CONSISTENCY: sum(individual)=%u >= all=%u (overlap=%u)",
287 (unsigned)sum_individual, (unsigned)all_count,
288 (unsigned)(sum_individual - all_count));
289 }
