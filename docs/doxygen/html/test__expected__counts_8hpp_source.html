332inline void validate_all_consistency() {
333 // Issue #173: no `all` class anymore. Every fine-grained class must
334 // be covered by at least one aggregate or by a scripted standalone
335 // special; `safe`/`selftest` are curated overlapping subsets and the
336 // foreign-arch classes are covered off-arch, so all four groups are
337 // excluded from the fine side of the comparison. Intentional overlap
338 // is allowed and reported: hal_apic + apic_tpr are members of both
339 // `hal` and `smp` (HAL devices AND SMP coordination primitives), and
340 // `testrunner` names both a fine class and its aggregate.
341 static constexpr const char *k_aggregates[] = {
342 "core", "ipc", "capability", "proc_elf",
343 "storage", "servers", "drivers", "hal",
344 "smp", "smp_multicpu", "deadline", "ui",
345 "logging_debug", "random", "bench", "testrunner",
346 nullptr,
347 };
348 static constexpr const char *k_specials[] = {
349 "ahci_live", "iommu_live", "task_fpu", "task_tcb_log",
350 nullptr,
351 };
352 size_t sum_covered = 0;
353 size_t sum_fine = 0;
354 size_t sum_singletons = 0; // testrunner fine + specials: counted
355 // once each on the fine side
356 for (size_t i = 0; i < k_expected_count_size; ++i) {
357 const char *name = k_expected_counts[i].name;
358 size_t c = arch_count(k_expected_counts[i]);
359 if (__builtin_strcmp(name, "safe") == 0 ||
360 __builtin_strcmp(name, "selftest") == 0 ||
361 __builtin_strcmp(name, "arch_aarch64") == 0 ||
362 __builtin_strcmp(name, "arch_riscv64") == 0) {
363 continue;
364 }
365 bool is_covered = false;
366 bool is_singleton = false;
367 for (const char *const *p = k_aggregates; *p != nullptr; ++p) {
368 if (__builtin_strcmp(name, *p) == 0) {
369 is_covered = true;
370 break;
371 }
372 }
373 if (!is_covered) {
374 for (const char *const *p = k_specials; *p != nullptr; ++p) {
375 if (__builtin_strcmp(name, *p) == 0) {
376 is_covered = true;
377 is_singleton = true;
378 break;
379 }
380 }
381 } else if (__builtin_strcmp(name, "testrunner") == 0) {
382 is_singleton = true;
383 }
384 if (is_covered) {
385 sum_covered += c;
386 } else {
387 sum_fine += c;
388 }
389 if (is_singleton) {
390 sum_singletons += c;
391 }
392 }
393 if (sum_covered < sum_fine + sum_singletons) {
394 Logger::warn("[TCOUNT] CONSISTENCY: aggregates+specials=%u < "
395 "fine+singletons=%u -- a fine class is uncovered",
396 (unsigned)sum_covered,
397 (unsigned)(sum_fine + sum_singletons));
398 } else {
399 Logger::info(
400 "[TCOUNT] CONSISTENCY: aggregates+specials=%u >= "
401 "fine+singletons=%u (overlap=%u, incl. hal_apic+apic_tpr in "
402 "hal+smp)",
403 (unsigned)sum_covered, (unsigned)(sum_fine + sum_singletons),
404 (unsigned)(sum_covered - sum_fine - sum_singletons));
405 }
406}
