45// ---------------------------------------------------------------------------
46// Architecture-specific PCI config space access
47// x86_64: CF8/CFC port I/O (0xCF8 / 0xCFC)
48// aarch64: ECAM memory-mapped (QEMU virt pcie-ecam at 0x4010000000,
49// issue #216)
50// riscv64: ECAM memory-mapped (QEMU virt pcie-ecam at 0x30000000,
51// issue #29)
52//
53// All access functions work on dword-aligned addresses internally, matching
54// the x86 CF8/CFC semantics where bits [1:0] of the register offset select
55// the byte/word within the dword. This ensures identical behavior across
56// all architectures.
57// ---------------------------------------------------------------------------

79constexpr uint16_t PCI_CONFIG_ADDR = 0xCF8;
81constexpr uint16_t PCI_CONFIG_DATA = 0xCFC;
82
88inline uint64_t pci_make_addr(PciBdf bdf, uint8_t reg) {
89 return 0x80000000U | (static_cast<uint32_t>(bdf.bus) << 16) |
90 (static_cast<uint32_t>(bdf.device) << 11) |
91 (static_cast<uint32_t>(bdf.function) << 8) | reg;
92}
93
97inline uint32_t pci_config_readl(uint64_t addr) {
98 outl(PCI_CONFIG_ADDR, static_cast<uint32_t>(addr & ~3ULL));
99 return inl(PCI_CONFIG_DATA);
100}
101
105inline uint8_t pci_config_readb(uint64_t addr) {
106 uint32_t val = pci_config_readl(addr);
107 return static_cast<uint8_t>((val >> ((addr & 3) * 8)) & 0xFF);
108}
109
113inline uint16_t pci_config_readw(uint64_t addr) {
114 uint32_t val = pci_config_readl(addr);
115 return static_cast<uint16_t>((val >> ((addr & 2) * 8)) & 0xFFFF);
116}
117
121inline void pci_config_writel(uint64_t addr, uint32_t val) {
122 outl(PCI_CONFIG_ADDR, static_cast<uint32_t>(addr & ~3ULL));
123 outl(PCI_CONFIG_DATA, val);
124}
125
130inline void pci_config_writew(uint64_t addr, uint16_t val) {
131 uint32_t old = pci_config_readl(addr);
132 uint32_t shift = (addr & 2) * 8;
133 uint32_t new_val =
134 (old & ~(0xFFFF << shift)) | (static_cast<uint32_t>(val) << shift);
135 pci_config_writel(addr, new_val);
136}
