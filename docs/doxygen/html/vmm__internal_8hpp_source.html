54// Test-time kernel-VA touch notifiers (issue #254 H2: eleven
55// is_test_active() control-flow branches hoisted here, bit-identical).
56// WHY the flags exist: test tasks legitimately map/unmap kernel-half VAs;
57// the snapshot mechanism must know which half was touched so restore rewinds
58// exactly that half (HHDM PD / identity PD / Sv39 L1). Release stores pair
59// with the acquire take in snapshot_restore (issues #60/#152). WHY the
60// warns stay: silent kernel-half mutation hides test pollution; each message
61// names the operation + index domain so the polluting test is identifiable.
62enum class TestTouchWarn {
63 kMapL1,
64 kMapPd,
65 kUnmapL0,
66 kUnmapPml4,
67 kAccessPml4,
68};

79// Page-offset / block-base masks (issue #363 checkbox 2, #365 1 GiB):
80// single named definitions replacing inline 0xFFF/0x1FFFFF/0x3FFFFFFF
81// literals. All arches use 4 KiB base pages; x86_64/aarch64 split 2 MiB
82// huge leaves, the PDPT free path reasons about 1 GiB block bases.
83inline constexpr uint64_t kPageOffsetMask = 0xFFFULL;
84inline constexpr uint64_t kHuge2MOffsetMask = 0x1FFFFFULL;
85inline constexpr uint64_t kHuge2MBaseMask = ~0x1FFFFFULL;
86static_assert(kPageOffsetMask + 1 == CONFIG_PAGE_SIZE,
87 "page offset mask must match the base page size");
88static_assert(kHuge2MOffsetMask == 512 * CONFIG_PAGE_SIZE - 1,
89 "2M offset mask must cover exactly one huge leaf");
90static_assert(kHuge2MBaseMask == ~kHuge2MOffsetMask,
91 "huge base mask must be the offset mask complement");
92// 1 GiB block geometry (issue #365): the only multi-megabyte level the
93// teardown free path reasons about. x86_64 PDPT and aarch64 L1 blocks
94// share this level; riscv64 has no 1 GiB user leaf, only the defensive
95// L0 site. Decoded bases come from the per-arch frame codec
96// (vmm_arch_block_base / sv39_pte_phys) — never a raw PTE mask.
97inline constexpr uint64_t kBlock1GOffsetMask = 0x3FFFFFFFULL;
98inline constexpr uint64_t kBlock1GBaseMask = ~kBlock1GOffsetMask;
99static_assert(kBlock1GOffsetMask + 1 == 512ULL * (kHuge2MOffsetMask + 1),
100 "1 GiB block must be exactly 512 x 2 MiB");
101static_assert(kBlock1GBaseMask == ~kBlock1GOffsetMask,
102 "1 GiB base mask must be the offset complement");

110[[nodiscard]] constexpr bool decode_block_base(uint64_t phys,
111 uint64_t offset_mask,
112 uint64_t &out_base) noexcept {
113 if ((phys & offset_mask) != 0)
114 return false;
115 out_base = phys & ~offset_mask;
116 return true;
117}

125// Issue #152: Sv39 L0 slot covered by the snapshot backend. The HHDM RAM
126// window (boot.S: 128 × 2MB leaves for PA 0x80000000-0x8FFFFFFF) lives
127// under L0[258] — the only HHDM L1 tests can split (L0[256/257] map
128// phys below RAM and have no block leaves). Gating is exact-match so the
129// armed flag always has a restore covering it (no silent holes). There
130// is deliberately no identity slot: the boot identity map (L0[2]) is
131// dead at runtime (L0[2] reads zero; the kernel runs via HHDM).
132inline constexpr size_t kHhdmL0Idx = 258;
133static_assert(
134 ((CONFIG_HHDM_OFFSET + 0x80000000ULL) >> 30 & VMM::TABLE_MASK) ==
135 kHhdmL0Idx,
