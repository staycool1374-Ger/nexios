54// Test-time kernel-VA touch notifiers (issue #254 H2: eleven
55// is_test_active() control-flow branches hoisted here, bit-identical).
56// WHY the flags exist: test tasks legitimately map/unmap kernel-half VAs;
57// the snapshot mechanism must know which half was touched so restore rewinds
58// exactly that half (HHDM PD / identity PD / Sv39 L1). Release stores pair
59// with the acquire take in snapshot_restore (issues #60/#152). WHY the
60// warns stay: silent kernel-half mutation hides test pollution; each message
61// names the operation + index domain so the polluting test is identifiable.
62enum class TestTouchWarn {
63 kMapL1,
64 kMapPd,
65 kUnmapL0,
66 kUnmapPml4,
67 kAccessPml4,
68};

79// Page-offset / block-base masks (issue #363 checkbox 2): single named
80// definitions replacing inline 0xFFF/0x1FFFFF/0x3FFFFF literals. All arches
81// use 4 KiB base pages; x86_64/aarch64 split 2 MiB huge leaves, the PDPT
82// free path reasons about 1 GiB block bases.
83inline constexpr uint64_t kPageOffsetMask = 0xFFFULL;
84inline constexpr uint64_t kHuge2MOffsetMask = 0x1FFFFFULL;
85inline constexpr uint64_t kHuge2MBaseMask = ~0x1FFFFFULL;
86inline constexpr uint64_t kPdptBlockBaseMask = ~0x3FFFFFULL;
87static_assert(kPageOffsetMask + 1 == CONFIG_PAGE_SIZE,
88 "page offset mask must match the base page size");
89static_assert(kHuge2MOffsetMask == 512 * CONFIG_PAGE_SIZE - 1,
90 "2M offset mask must cover exactly one huge leaf");
91static_assert(kHuge2MBaseMask == ~kHuge2MOffsetMask,
92 "huge base mask must be the offset mask complement");
93// NOTE: kPdptBlockBaseMask preserves the pre-split 22-bit shape verbatim
94// (free_user_pages PDPT-block base). It is deliberately NOT re-derived
95// from 1 GiB geometry here — any reinterpretation needs an owner decision
96// (issue #363 follow-up), so no static_assert pins a geometry claim.

103#if defined(CONFIG_ARCH_RISCV64)
104// Issue #152: Sv39 L0 slot covered by the snapshot backend. The HHDM RAM
105// window (boot.S: 128 × 2MB leaves for PA 0x80000000-0x8FFFFFFF) lives
106// under L0[258] — the only HHDM L1 tests can split (L0[256/257] map
107// phys below RAM and have no block leaves). Gating is exact-match so the
108// armed flag always has a restore covering it (no silent holes). There
109// is deliberately no identity slot: the boot identity map (L0[2]) is
110// dead at runtime (L0[2] reads zero; the kernel runs via HHDM).
111inline constexpr size_t kHhdmL0Idx = 258;
112static_assert(
113 ((CONFIG_HHDM_OFFSET + 0x80000000ULL) >> 30 & VMM::TABLE_MASK) ==
114 kHhdmL0Idx,
115 "HHDM RAM L0 moved");
