116template <typename T>
117bool verify_and_write(T &target, const T &value, WriteClass cls,
118 const WriteContext &ctx, const char *name) noexcept {
119 bool accepted = true;
120 switch (cls) {
121 case WriteClass::IDEMPOTENT:
122 // A bool already true (or any same-value write) need not be repeated.
123 if (target == value)
124 accepted = false;
125 break;
126 case WriteClass::BOOT_ONLY:
127 if (ctx.phase != StatePhase::BOOT)
128 accepted = false;
129 break;
130 case WriteClass::NEVER_WRITE:
131 accepted = false;
132 break;
133 default:
134 break;
135 }
136 if (accepted)
137 target = value;
138#ifdef CONFIG_DEBUG
139 audit_write(name, accepted);
140#else
141 (void)name;
142#endif
143 return accepted;
144}
