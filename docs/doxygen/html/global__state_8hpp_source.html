100template <typename T>
101bool verify_and_write(T &target, const T &value, WriteClass cls,
102 const WriteContext &ctx, const char *name) noexcept {
103 bool accepted = true;
104 switch (cls) {
105 case WriteClass::IDEMPOTENT:
106 // A bool already true (or any same-value write) need not be repeated.
107 if (target == value)
108 accepted = false;
109 break;
110 case WriteClass::BOOT_ONLY:
111 if (ctx.phase != StatePhase::BOOT)
112 accepted = false;
113 break;
114 case WriteClass::NEVER_WRITE:
115 accepted = false;
116 break;
117 default:
118 break;
119 }
120 if (accepted)
121 target = value;
122#ifdef CONFIG_DEBUG
123 audit_write(name, accepted);
124#else
125 (void)name;
126#endif
127 return accepted;
128}
