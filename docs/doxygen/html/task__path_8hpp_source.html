132// Non-zero and pairwise-distinct from the empty chain — a fold-value typo
133// or a broken mix would collapse these.
134static_assert(k_task_create_expected != 0);
135static_assert(k_task_cleanup_expected != k_task_create_expected);
136static_assert(k_task_cleanup_pinned_expected != k_task_create_expected);
137static_assert(k_task_cleanup_pinned_expected != k_task_cleanup_expected);
138// Teardown chains extend the cleanup fold with a scheduler prefix —
139// a fold-value typo or a broken mix would collapse these.
140static_assert(k_teardown_canonical_expected != k_task_cleanup_expected);
141static_assert(k_teardown_reap_expected != k_task_cleanup_expected);
142static_assert(k_teardown_canonical_expected != k_teardown_reap_expected);
