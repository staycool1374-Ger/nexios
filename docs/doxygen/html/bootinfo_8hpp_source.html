61 void add_region(uint64_t base, uint64_t size, uint32_t type) {
62 if (num_mem_regions < BOOTINFO_MAX_MEM_REGIONS) {
63 mem_regions[num_mem_regions].base = base;
64 mem_regions[num_mem_regions].size = size;
65 mem_regions[num_mem_regions].type = type;
