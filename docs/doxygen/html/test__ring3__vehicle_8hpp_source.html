70inline uint64_t write_data_file(const char *path, const uint8_t *data,
71 size_t len) {
72 kernel::test::mark_vfs_touched();
73 if (vfs::create(path, 0) != 0)
74 return 0;
75 vfs::Vnode *file = vfs::resolve(path);
76 if (file == nullptr || file->ops == nullptr ||
77 file->ops->write == nullptr) {
78 unlink_path(path);
79 return 0;
80 }
81 size_t off = 0;
82 while (off < len) {
83 size_t chunk = len - off;
84 if (chunk > 4096)
85 chunk = 4096;
86 int64_t written =
87 file->ops->write(*file, data + off, chunk, off);
88 if (written <= 0) {
89 unlink_path(path);
90 return 0;
91 }
92 off += static_cast<size_t>(written);
93 }
94 return (off == len) ? len : 0ULL;
95}

119// Parse `wanted` unsigned decimals after a tag into out; false on miss.
120// (Whitespace-tolerant: skips spaces AND newlines between numbers so
121// multi-line marker blocks parse.)
122inline bool parse_tag_line(const char *buf, const char *tag, uint64_t *out,
123 int wanted, size_t cap) {
124 size_t n = 0;
125 while (buf[n] != '\0' && n < cap) {
126 size_t j = 0;
127 while (tag[j] != '\0' && buf[n + j] == tag[j])
128 ++j;
129 if (tag[j] == '\0') {
130 size_t p = n + j;
131 for (int k = 0; k < wanted; ++k) {
132 while ((buf[p] == ' ' || buf[p] == '\n') && p < cap)
133 ++p;
134 if (p >= cap || buf[p] < '0' || buf[p] > '9')
135 return false;
136 uint64_t v = 0;
137 while (p < cap && buf[p] >= '0' && buf[p] <= '9') {
138 v = v * 10 + static_cast<uint64_t>(buf[p] - '0');
139 ++p;
140 }
141 out[k] = v;
142 }
143 return true;
144 }
145 ++n;
146 }
147 return false;
148}

150// Run the image; stage_extra (optional) stages per-test files after the
151// standard three and before load; cleanup_extra unlinks them after the
152// capture is read. Asserts happen after teardown (allocation-free).
153inline Outcome run(void (*stage_extra)() = nullptr,
154 void (*cleanup_extra)() = nullptr, char *capture = nullptr,
155 size_t capture_size = 0) {
156 Outcome out{};
157 elf::ElfLoader::reset();
158 if (write_image_file() == 0)
159 return out;
160 if (write_data_file(kStdinPath,
161 reinterpret_cast<const uint8_t *>(kStdinData),
162 5) != 5) {
163 unlink_path(kImagePath);
164 return out;
165 }
166 kernel::test::mark_vfs_touched();
167 if (vfs::create(kStdoutPath, 0) != 0) {
168 unlink_path(kImagePath);
169 unlink_path(kStdinPath);
170 return out;
171 }
172 if (stage_extra != nullptr)
173 stage_extra();
174 if (elf::ElfLoader::request_load(kImagePath) != elf::LoadResult::OK) {
175 unlink_path(kImagePath);
176 unlink_path(kStdinPath);
177 unlink_path(kStdoutPath);
178 if (cleanup_extra != nullptr)
179 cleanup_extra();
180 return out;
181 }
182 elf::ElfLoader::wait_loader_idle();
183 TaskControlBlock *t = elf::ElfLoader::take_completed();
184 if (t != nullptr && t->get_page_table() != 0) {
185 uint64_t p_text =
186 VMM::virt_to_phys_in_pml4(0x400000, t->get_page_table());
187 uint64_t p_stack = VMM::virt_to_phys_in_pml4(
188 mem::STACK_VADDR + 4096, t->get_page_table());
189 if (p_text == 0 || p_stack == 0) {
190 elf::ElfLoader::destroy_completed_tcb(t);
191 unlink_path(kImagePath);
192 unlink_path(kStdinPath);
193 unlink_path(kStdoutPath);
194 if (cleanup_extra != nullptr)
195 cleanup_extra();
196 return out;
197 }
198 }
199 if (t == nullptr || t->get_page_table() == 0 || !t->is_user_) {
200 if (t != nullptr)
201 elf::ElfLoader::destroy_completed_tcb(t);
202 unlink_path(kImagePath);
203 unlink_path(kStdinPath);
204 unlink_path(kStdoutPath);
205 if (cleanup_extra != nullptr)
206 cleanup_extra();
207 return out;
208 }
209 // Repoint fd 0 at the tmpfs stdin file (direct slot overwrite:
210 // FdTable::free decs a vnode ref never inc'd — destroy_completed_tcb
211 // no-dec discipline).
212 if (!t->fd_table.fds[0].used) {
213 elf::ElfLoader::destroy_completed_tcb(t);
214 unlink_path(kImagePath);
215 unlink_path(kStdinPath);
216 unlink_path(kStdoutPath);
217 if (cleanup_extra != nullptr)
218 cleanup_extra();
219 return out;
220 }
221 vfs::Vnode *stdin_vnode = vfs::resolve(kStdinPath);
222 if (stdin_vnode == nullptr) {
223 elf::ElfLoader::destroy_completed_tcb(t);
224 unlink_path(kImagePath);
225 unlink_path(kStdinPath);
226 unlink_path(kStdoutPath);
227 if (cleanup_extra != nullptr)
228 cleanup_extra();
229 return out;
230 }
231 t->fd_table.fds[0].vnode = stdin_vnode;
232 t->fd_table.fds[0].offset = 0;
233 t->fd_table.fds[0].flags = 0;
234
235 // Repoint stdout at the tmpfs capture file (same discipline).
236 if (!t->fd_table.fds[1].used) {
237 elf::ElfLoader::destroy_completed_tcb(t);
238 unlink_path(kImagePath);
239 unlink_path(kStdinPath);
240 unlink_path(kStdoutPath);
241 if (cleanup_extra != nullptr)
242 cleanup_extra();
243 return out;
244 }
245 vfs::Vnode *stdout_vnode = vfs::resolve(kStdoutPath);
246 if (stdout_vnode == nullptr) {
247 elf::ElfLoader::destroy_completed_tcb(t);
248 unlink_path(kImagePath);
249 unlink_path(kStdinPath);
250 unlink_path(kStdoutPath);
251 if (cleanup_extra != nullptr)
252 cleanup_extra();
253 return out;
254 }
255 t->fd_table.fds[1].vnode = stdout_vnode;
256 t->fd_table.fds[1].offset = 0;
257 t->fd_table.fds[1].flags = 0;
258
259 // Prio 11: the harness (prio 10) would starve a lower-priority waitee
260 // after the first dispatch demotes it to READY. Set before add_task.
261 t->priority = 11;
262 t->base_priority = 11;
263 {
264 arch::IrqGuard ig{};
265 Scheduler::add_task(*t);
266 }
267 Scheduler::reschedule();
268 for (uint64_t i = 0;
269 i < kJoinSpins && t->state != TaskState::TERMINATED; ++i)
270 arch::pause();
271 bool exited = (t->state == TaskState::TERMINATED);
272 bool clean = exited && (t->exit_code == 0);
273 out.program_break = t->program_break;
274 // Read back the capture BEFORE assert (allocation-free teardown rule).
275 if (capture != nullptr && capture_size > 1) {
276 vfs::Vnode *cap = vfs::resolve(kStdoutPath);
277 size_t pos = 0;
278 if (cap != nullptr && cap->ops != nullptr &&
279 cap->ops->read != nullptr) {
280 while (pos < capture_size - 1) {
281 int64_t rd = cap->ops->read(
282 *cap, reinterpret_cast<uint8_t *>(capture + pos), 1,
283 pos);
284 if (rd <= 0)
285 break;
286 pos += static_cast<size_t>(rd);
287 }
288 }
289 capture[pos] = '\0';
290 }
291 if (TaskControlBlock::is_valid(t) &&
292 (t->state != TaskState::TERMINATED || t->exit_code == 0))
293 Scheduler::terminate(*t, 0);
294 Scheduler::drain_zombie_list();
295 unlink_path(kImagePath);
296 unlink_path(kStdinPath);
297 unlink_path(kStdoutPath);
298 if (cleanup_extra != nullptr)
299 cleanup_extra();
300 out.joined = clean;
301 out.clean_exit = clean;
302 (void)exited;
303 return out;
304}

310// Read a whole tmpfs file into buf (up to size-1 bytes, NUL-terminated).
311// Returns bytes read, or -1 on any failure. Marks VFS touched.
312inline int64_t read_file(const char *path, char *buf, size_t size) {
313 if (size == 0)
314 return -1;
315 kernel::test::mark_vfs_touched();
316 vfs::Vnode *file = vfs::resolve(path);
317 if (file == nullptr || file->ops == nullptr ||
318 file->ops->read == nullptr)
319 return -1;
320 size_t pos = 0;
321 while (pos < size - 1) {
322 int64_t rd =
323 file->ops->read(*file, reinterpret_cast<uint8_t *>(buf + pos),
324 size - 1 - pos, pos);
325 if (rd <= 0)
326 break;
327 pos += static_cast<size_t>(rd);
328 }
329 buf[pos] = '\0';
330 return static_cast<int64_t>(pos);
331}
