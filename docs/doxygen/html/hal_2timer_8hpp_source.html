46inline uint64_t monotonic_commit(uint64_t* cell, uint64_t now) {
47 uint64_t prev = __atomic_load_n(cell, __ATOMIC_ACQUIRE);
48 while (now > prev) {
49 uint64_t expect = prev;
50 const bool swapped = __atomic_compare_exchange_n(
51 cell, &expect, now, false, __ATOMIC_ACQ_REL,
52 __ATOMIC_ACQUIRE);
53 if (swapped) {
54 prev = now;
55 break;
56 }
57 prev = expect;
58 }
59 return (now > prev) ? now : prev;
60}
