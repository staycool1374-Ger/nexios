44class [[nodiscard]] TprGuard {
45 public:
49 explicit TprGuard(uint8_t cls) noexcept : saved_(current_class()) {
50#if defined(CONFIG_ARCH_X86_64)
51 // INV-TPR2 enforced: the raise/shadow pair is only atomic vs
52 // interrupts under IF=0. TPR narrows which vectors may preempt;
53 // it never replaces the caller's lock + cli discipline.
54 ENSURE(!interrupts_enabled());
55#endif
56 raise(cls);
57 }
58
60 ~TprGuard() noexcept {
61 restore(saved_);
62 }
63
64 TprGuard(const TprGuard &) = delete;
65 TprGuard &operator=(const TprGuard &) = delete;
66 TprGuard(TprGuard &&) = delete;
67 TprGuard &operator=(TprGuard &&) = delete;
68
69 private:
71 uint8_t saved_;
72
74 static uint8_t current_class() noexcept {
75#if defined(CONFIG_ARCH_X86_64)
76 return static_cast<uint8_t>(
77 per_cpu_current()->tpr_shadow & 0xF0U);
78#else
79 return 0;
80#endif
81 }
82
84 static void raise(uint8_t cls) noexcept {
85#if defined(CONFIG_ARCH_X86_64)
86 APIC::tpr_raise(cls);
87 per_cpu_current()->tpr_shadow =
88 static_cast<uint64_t>(APIC::get_tpr_class());
89#else
90 (void)cls;
91#endif
92 }
93
95 static void restore(uint8_t cls) noexcept {
96#if defined(CONFIG_ARCH_X86_64)
97 APIC::tpr_restore(cls);
98 per_cpu_current()->tpr_shadow =
99 static_cast<uint64_t>(APIC::get_tpr_class());
100#else
101 (void)cls;
102#endif
103 }
104};
