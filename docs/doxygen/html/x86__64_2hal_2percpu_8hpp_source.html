47struct alignas(arch::PAGE_SIZE) PerCpu {
48 // ─── ABI-frozen slots (existing gs:0x00/0x08 semantics) ─────────────────
49 uint64_t user_rsp; // gs:0x00 — saved user RSP (syscall entry)
50 uint64_t kernel_rsp; // gs:0x08 — loaded kernel stack (ISR epilogue)
51
52 // ─── CPU identity ──────────────────────────────────────────────────────
53 uint64_t cpu_id; // gs:0x10 — logical CPU index (0..CONFIG_MAX_CPUS-1)
54 uint64_t lapic_id; // gs:0x18 — LAPIC ID from MADT
55
56 // ─── Migrated from globals (issue #94) ──────────────────────────────────
57 uint64_t isr_nesting_depth; // gs:0x20 — was global isr_nesting_depth
58 uint64_t irq_entry_tsc; // gs:0x28 — was global irq_entry_tsc
59
60 // ─── FPU / scheduler ───────────────────────────────────────────────────
61 // Own-CPU FPU register owner (issue #151): which task's FPU state is
62 // currently live in this CPU's registers. nullptr = none since boot.
63 kernel::TaskControlBlock *fpu_owner; // gs:0x30 — per-CPU (was global)
64 void *current_task; // gs:0x38 — RESERVED (superseded by CpuContext
65 // array, issue #25 C1; kept for layout)
66
67 // ─── Interrupt prioritization (issue #26) ──────────────────────────────
68 // Own-CPU intended TPR class (source of truth; HW re-asserted from it).
69 // C++-only via per_cpu_current()/cpu_index() — no asm access, no alias.
70 uint64_t tpr_shadow; // gs:0x40 — default TPR_CLASS_ACCEPT_ALL
71
72 // ─── Deferred-switch atoms: NOT in this page ─────────────────────────────
73 // C1 stores them as Scheduler/global_state arrays indexed by CPU
74 // (arch-neutral C++; riscv asm keeps working via array base == [0]).
75 // x86 asm indexes via gs:0x10 (INV-PC4 forbids BARE [rel scheduler_*]).
76
77 // ─── Reserved / future expansion ───────────────────────────────────────
78 // NOTE: explicit fields total 483 u64s; alignas(PAGE_SIZE) tail-pads
79 // sizeof to 4096 (enforced by the static_assert below). Keep the
80 // explicit total constant when adding fields (shrink reserved).
81 uint64_t reserved[474];
82};

84static_assert(sizeof(PerCpu) == arch::PAGE_SIZE,
85 "PerCpu must be exactly one 4 KiB page");

89extern PerCpu per_cpu[CONFIG_MAX_CPUS];

95inline PerCpu *per_cpu_current() {
96 uint64_t base = arch::rdmsr(arch::MSR_GS_BASE);
97 uint64_t lo = reinterpret_cast<uint64_t>(&per_cpu[0]);
98 if (base >= lo && base < lo + sizeof(per_cpu))
99 return &per_cpu[(base - lo) / arch::PAGE_SIZE];
100 return &per_cpu[0];
101}
102
104inline uint64_t cpu_id() {
105 return per_cpu_current()->cpu_id;
106}
107
110inline uint64_t cpu_index() {
111 return static_cast<uint64_t>(per_cpu_current() - &per_cpu[0]);
112}
113
115inline uint64_t lapic_id() {
116 return per_cpu_current()->lapic_id;
117}
