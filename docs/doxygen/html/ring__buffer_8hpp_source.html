31class RingBuffer {
32 public:
33 static constexpr size_t BUFFER_SIZE = 32768;
34
36 void putchar(char c) {
37 size_t w = write_pos_;
38 size_t next = (w + 1) % BUFFER_SIZE;
39 if (next == atomic_load(&read_pos_, __ATOMIC_ACQUIRE))
40 return;
41 buf_[w] = c;
42 atomic_store(&write_pos_, next, __ATOMIC_RELEASE);
43 }
44
46 void puts(const char *s) {
47 while (*s)
48 putchar(*s++);
49 }
50
53 size_t read(char *dst, size_t size) {
54 size_t r = atomic_load(&read_pos_, __ATOMIC_RELAXED);
55 size_t w = atomic_load(&write_pos_, __ATOMIC_ACQUIRE);
56 size_t written = 0;
57 while (r != w && written < size) {
58 *dst++ = buf_[r];
59 r = (r + 1) % BUFFER_SIZE;
60 ++written;
61 }
62 atomic_store(&read_pos_, r, __ATOMIC_RELEASE);
63 return written;
64 }
65
67 void clear() {
68 atomic_store(&read_pos_, atomic_load(&write_pos_, __ATOMIC_RELAXED),
69 __ATOMIC_RELEASE);
70 }
71
73 bool empty() const {
74 return atomic_load(&read_pos_, __ATOMIC_ACQUIRE) ==
75 atomic_load(&write_pos_, __ATOMIC_ACQUIRE);
76 }
77
78 RingBuffer() : buf_{} {
79 }
80
81 private:
82 char buf_[BUFFER_SIZE];
83 alignas(64) volatile size_t write_pos_ = 0;
84 alignas(64) volatile size_t read_pos_ = 0;
85};
