174inline uint64_t gicr_frame_for_aff0(uint64_t aff0) {
175 for (uint64_t idx = 0; idx < CONFIG_MAX_CPUS; ++idx) {
176 volatile uint64_t *typer = reinterpret_cast<volatile uint64_t *>(
177 arch::HHDM_OFFSET + GICR_RD_BASE + idx * GICR_STRIDE + GICR_TYPER);
178 const uint64_t typer_val = *typer;
179 if (((typer_val >> 32U) & 0xFFU) == (aff0 & 0xFFU)) {
180 return arch::HHDM_OFFSET + GICR_RD_BASE + idx * GICR_STRIDE;
181 }
182 }
183 return 0;
184}

226inline bool gic_send_sgi(uint32_t target_cpu, uint32_t intid) {
227 if (target_cpu >= CONFIG_MAX_CPUS || intid > 15U) {
228 return false;
229 }
230 // Only the kernel SGI set may be programmed (INTIDs 8/9/10 land on
231 // free IDT vectors; 0/1/2 would alias TIMER/KEYBOARD/SYSCALL).
232 if (intid != SMP_LOOPBACK_SGI && intid != SMP_SCHED_SGI &&
233 intid != SMP_SHOOTDOWN_SGI) {
234 return false;
235 }
236 if (gic_version() >= 3U) {
237 const uint64_t sgi_value =
238 (static_cast<uint64_t>(intid & 0xFU) << 24) |
239 (static_cast<uint64_t>(1U << target_cpu));
240 asm volatile("msr ICC_SGI1R_EL1, %0" : : "r"(sgi_value) : "memory");
241 asm volatile("isb");
242 } else {
243 const uint32_t sgir_value =
244 GICD_SGIR_FILTER_TARGET_LIST |
245 ((1U << target_cpu) << GICD_SGIR_TARGET_SHIFT) |
246 (intid & GICD_SGIR_INTID_MASK);
247 gicd_reg(GICD_SGIR)[0] = sgir_value;
248 dsb_sy();
249 }
250 return true;
251}

260inline void gic_ap_init() {
261 if (gic_version() >= 3U) {
262 const uint64_t aff0 = mpidr_aff0();
263 const uint64_t rd_frame = gicr_frame_for_aff0(aff0);
264 if (rd_frame != 0U) {
265 volatile uint32_t *rr =
266 reinterpret_cast<volatile uint32_t *>(rd_frame);
267 volatile uint32_t *rs = reinterpret_cast<volatile uint32_t *>(
268 rd_frame + (GICR_SGI_BASE - GICR_RD_BASE));
269 uint32_t waker = rr[GICR_WAKER / 4U];
270 waker &= ~(1U << 0U);
271 rr[GICR_WAKER / 4U] = waker;
272 dsb_sy();
273 for (uint32_t poll = 0; poll < GICR_WAKER_MAX_POLLS; ++poll) {
274 if ((rr[GICR_WAKER / 4U] & GICR_WAKER_CHILDREN_ONLINE) !=
275 0U) {
276 break;
277 }
278 dsb_sy();
279 }
280 rs[GICR_ISENABLER0 / 4U] |=
281 SMP_SGI_ENABLE_MASK | (1U << 30U);
282 dsb_sy();
283 }
284 gic_v3_set_sre(true);
285 gic_v3_set_pmr(0xFF);
286 gic_v3_set_igrpen1(true);
287 isb();
288 } else {
289 gicc_reg(GICC_PMR)[0] = 0xFF;
290 dsb_sy();
291 gicc_reg(GICC_CTLR)[0] = GICC_CTLR_ENABLE;
292 dsb_sy();
293 }
294}
