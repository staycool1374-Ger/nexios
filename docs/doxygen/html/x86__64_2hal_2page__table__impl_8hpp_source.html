91inline void tlb_purge_context(uint16_t pcid) {
92 TlbFlushStats &store = tlb_flush_stats_store();
93 if (has_invpcid()) {
94 __atomic_fetch_add(&store.ctx_invpcid, 1ULL, __ATOMIC_RELAXED);
95 InvpcidDesc desc = invpcid_build_desc(InvpcidType::SINGLE_CONTEXT,
96 pcid, 0);
97 invpcid_emit(InvpcidType::SINGLE_CONTEXT, desc);
98 } else {
99 __atomic_fetch_add(&store.ctx_fallback, 1ULL, __ATOMIC_RELAXED);
100 uint64_t cr3 = read_cr3();
101 write_cr3(cr3 & ~0xFFFULL);
102 }
103}
