205 compute_window_pages(uint64_t mem_size, uint64_t window_base) {
206 uint64_t total = mem_size / PAGE_SIZE;
207 uint64_t base = window_base / PAGE_SIZE;
208 uint64_t end = base + arch::HHDM_WINDOW_SIZE / PAGE_SIZE;
209 if (end > total) {
210 end = total;
211 }
212 if (base > end) {
213 base = end;
214 }
215 return WindowPages{base, end};
216 }
