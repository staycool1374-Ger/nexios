525 struct TaskIter {
526 uint64_t idx;
527 TaskControlBlock *cur_;
528 explicit TaskIter(uint64_t start = 0)
529 : idx(start), cur_(Scheduler::all_tasks_.first_ptr()) {
530 for (uint64_t i = 0; i < start; ++i) {
531 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
532 if (!cur_)
533 break;
534 }
535 }
536 TaskControlBlock *next(TaskControlBlock *exclude = nullptr) {
537 while (cur_) {
538 auto *t = cur_;
539 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
540 ++idx;
541 if (t && t->magic == TaskControlBlock::TCB_MAGIC &&
542 t != exclude)
543 return t;
544 }
545 return nullptr;
546 }
547 };

776 static bool is_queued_on(const TaskControlBlock &t, uint64_t cpu) noexcept {
777 if (t.in_ready_queue_) {
778 const ReadyQueueManager &rq = ready_queues_[cpu % CONFIG_MAX_CPUS];
779 for (uint64_t p = 0; p <= CONFIG_PRIORITY_CEILING; ++p) {
780 if (rq.queue(p).contains(t))
781 return true;
782 }
783 return false;
784 }
785 if (t.in_edf_queue_)
786 return queue_target(t) == (cpu % CONFIG_MAX_CPUS);
787 return false;
