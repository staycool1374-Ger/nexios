359 struct TaskIter {
360 uint64_t idx;
361 TaskControlBlock *cur_;
362 explicit TaskIter(uint64_t start = 0)
363 : idx(start), cur_(Scheduler::all_tasks_.first_ptr()) {
364 for (uint64_t i = 0; i < start; ++i) {
365 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
366 if (!cur_)
367 break;
368 }
369 }
370 TaskControlBlock *next(TaskControlBlock *exclude = nullptr) {
371 while (cur_) {
372 auto *t = cur_;
373 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
374 ++idx;
375 if (t && t->magic == TaskControlBlock::TCB_MAGIC &&
376 t != exclude)
377 return t;
378 }
379 return nullptr;
380 }
381 };
