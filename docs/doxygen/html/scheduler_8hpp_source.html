604 struct TaskIter {
605 uint64_t idx;
606 TaskControlBlock *cur_;
607 explicit TaskIter(uint64_t start = 0)
608 : idx(start), cur_(Scheduler::all_tasks_.first_ptr()) {
609 for (uint64_t i = 0; i < start; ++i) {
610 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
611 if (!cur_)
612 break;
613 }
614 }
615 TaskControlBlock *next(TaskControlBlock *exclude = nullptr) {
616 while (cur_) {
617 auto *t = cur_;
618 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
619 ++idx;
620 if (t && t->magic == TaskControlBlock::TCB_MAGIC &&
621 t != exclude)
622 return t;
623 }
624 return nullptr;
625 }
626 };

841 static uint64_t loadavg_step(uint64_t avg, uint64_t instant,
842 uint64_t window, uint64_t scale) noexcept {
843 if (window == 0 || scale == 0)
844 return 0;
845 if (avg > scale)
846 avg = scale;
847 if (instant > scale)
848 instant = scale;
849 int64_t diff = static_cast<int64_t>(instant) -
850 static_cast<int64_t>(avg);
851 int64_t step = diff / static_cast<int64_t>(window);
852 int64_t updated = static_cast<int64_t>(avg) + step;
853 if (updated < 0)
854 updated = 0;
855 if (updated > static_cast<int64_t>(scale))
856 updated = static_cast<int64_t>(scale);
857 return static_cast<uint64_t>(updated);
858 }
