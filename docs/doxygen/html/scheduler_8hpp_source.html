357 struct TaskIter {
358 uint64_t idx;
359 TaskControlBlock *cur_;
360 explicit TaskIter(uint64_t start = 0)
361 : idx(start), cur_(Scheduler::all_tasks_.first_ptr()) {
362 for (uint64_t i = 0; i < start; ++i) {
363 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
364 if (!cur_)
365 break;
366 }
367 }
368 TaskControlBlock *next(TaskControlBlock *exclude = nullptr) {
369 while (cur_) {
370 auto *t = cur_;
371 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
372 ++idx;
373 if (t && t->magic == TaskControlBlock::TCB_MAGIC &&
374 t != exclude)
375 return t;
376 }
377 return nullptr;
378 }
379 };
