449 struct TaskIter {
450 uint64_t idx;
451 TaskControlBlock *cur_;
452 explicit TaskIter(uint64_t start = 0)
453 : idx(start), cur_(Scheduler::all_tasks_.first_ptr()) {
454 for (uint64_t i = 0; i < start; ++i) {
455 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
456 if (!cur_)
457 break;
458 }
459 }
460 TaskControlBlock *next(TaskControlBlock *exclude = nullptr) {
461 while (cur_) {
462 auto *t = cur_;
463 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
464 ++idx;
465 if (t && t->magic == TaskControlBlock::TCB_MAGIC &&
466 t != exclude)
467 return t;
468 }
469 return nullptr;
470 }
471 };
