685 struct TaskIter {
686 uint64_t idx;
687 TaskControlBlock *cur_;
688 explicit TaskIter(uint64_t start = 0)
689 : idx(start), cur_(Scheduler::all_tasks_.first_ptr()) {
690 for (uint64_t i = 0; i < start; ++i) {
691 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
692 if (!cur_)
693 break;
694 }
695 }
696 TaskControlBlock *next(TaskControlBlock *exclude = nullptr) {
697 while (cur_) {
698 auto *t = cur_;
699 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
700 ++idx;
701 if (t && t->magic == TaskControlBlock::TCB_MAGIC &&
702 t != exclude)
703 return t;
704 }
705 return nullptr;
706 }
707 };

934 static uint64_t loadavg_step(uint64_t avg, uint64_t instant,
935 uint64_t window, uint64_t scale) noexcept {
936 if (window == 0 || scale == 0)
937 return 0;
938 if (avg > scale)
939 avg = scale;
940 if (instant > scale)
941 instant = scale;
942 int64_t diff = static_cast<int64_t>(instant) -
943 static_cast<int64_t>(avg);
944 int64_t step = diff / static_cast<int64_t>(window);
945 int64_t updated = static_cast<int64_t>(avg) + step;
946 if (updated < 0)
947 updated = 0;
948 if (updated > static_cast<int64_t>(scale))
949 updated = static_cast<int64_t>(scale);
950 return static_cast<uint64_t>(updated);
951 }
