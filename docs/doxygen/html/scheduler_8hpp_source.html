546 struct TaskIter {
547 uint64_t idx;
548 TaskControlBlock *cur_;
549 explicit TaskIter(uint64_t start = 0)
550 : idx(start), cur_(Scheduler::all_tasks_.first_ptr()) {
551 for (uint64_t i = 0; i < start; ++i) {
552 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
553 if (!cur_)
554 break;
555 }
556 }
557 TaskControlBlock *next(TaskControlBlock *exclude = nullptr) {
558 while (cur_) {
559 auto *t = cur_;
560 cur_ = Scheduler::all_tasks_.next_ptr(cur_);
561 ++idx;
562 if (t && t->magic == TaskControlBlock::TCB_MAGIC &&
563 t != exclude)
564 return t;
565 }
566 return nullptr;
567 }
568 };

781 static uint64_t loadavg_step(uint64_t avg, uint64_t instant,
782 uint64_t window, uint64_t scale) noexcept {
783 if (window == 0 || scale == 0)
784 return 0;
785 if (avg > scale)
786 avg = scale;
787 if (instant > scale)
788 instant = scale;
789 int64_t diff = static_cast<int64_t>(instant) -
790 static_cast<int64_t>(avg);
791 int64_t step = diff / static_cast<int64_t>(window);
792 int64_t updated = static_cast<int64_t>(avg) + step;
793 if (updated < 0)
794 updated = 0;
795 if (updated > static_cast<int64_t>(scale))
796 updated = static_cast<int64_t>(scale);
797 return static_cast<uint64_t>(updated);
798 }

866 static bool is_queued_on(const TaskControlBlock &t, uint64_t cpu) noexcept {
867 if (t.in_ready_queue_) {
868 const ReadyQueueManager &rq = ready_queues_[cpu % CONFIG_MAX_CPUS];
869 for (uint64_t p = 0; p <= CONFIG_PRIORITY_CEILING; ++p) {
870 if (rq.queue(p).contains(t))
871 return true;
872 }
873 return false;
874 }
875 if (t.in_edf_queue_)
876 return queue_target(t) == (cpu % CONFIG_MAX_CPUS);
877 return false;
878 }
