62constexpr uint32_t k_fast_word_regs[IPC_FAST_PAYLOAD_BYTES / 8] = {4, 5,
63 7, 8,
64 9, 10};

69inline void fast_msg_to_regs(const Message &msg, uint64_t *regs) {
70 for (size_t w = 0; w < sizeof(k_fast_word_regs) /
71 sizeof(k_fast_word_regs[0]);
72 ++w) {
73 uint64_t word = 0;
74 size_t off = w * 8;
75 if (off < msg.data_size) {
76 size_t n = msg.data_size - off;
77 if (n > 8)
78 n = 8;
79 for (size_t b = 0; b < n; ++b)
80 word |= static_cast<uint64_t>(msg.data[off + b]) << (8 * b);
81 }
82 regs[k_fast_word_regs[w]] = word;
83 }
84}

89inline void fast_regs_to_msg(uint64_t *regs, size_t data_size, Message &msg) {
90 for (size_t w = 0; w < sizeof(k_fast_word_regs) /
91 sizeof(k_fast_word_regs[0]);
92 ++w) {
93 uint64_t word = regs[k_fast_word_regs[w]];
94 size_t off = w * 8;
95 if (off >= data_size)
96 break;
97 size_t n = data_size - off;
98 if (n > 8)
99 n = 8;
100 for (size_t b = 0; b < n; ++b)
101 msg.data[off + b] = static_cast<uint8_t>((word >> (8 * b)) & 0xFF);
102 }
103}
