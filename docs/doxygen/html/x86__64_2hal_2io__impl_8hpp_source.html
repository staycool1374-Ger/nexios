186inline uint64_t rdtsc() {
187 uint32_t tsc_low, tsc_high;
188 asm volatile("rdtsc" : "=a"(tsc_low), "=d"(tsc_high));
189 return (static_cast<uint64_t>(tsc_high) << 32) | tsc_low;
190}
