106static inline long __syscall5(long num, long a0, long a1, long a2, long a3) {
107 long ret;
108 // ABI contract (single source of truth, §2 binding):
109 // x86_64: number in rax, args in rbx/rcx/rdx/rsi (int $0x80)
110 // aarch64: number in x8, args in x0-x3 (NexIOS: x8=num, x0-x3=args)
111 // riscv64: number in a7, args in a0-a3 (ecall)
112 // The arch syscall entry stubs must match these register choices.
113 // §2 ERROR (binding): uint64_t ret; (long)ret < 0 &&
114 // (unsigned long)ret > -4096UL ⇒ errno = -(long)ret, return -1;
115 // -4095 maps to errno, -4097 is valid success; no handler returns
116 // a kernel pointer; reachable errnos per CODING_STYLE §5 (never
117 // ENSURE). FAST rsi/rdi/r8-r11 layout (ipc-fastpath §3.2) is a
118 // separate path and out of scope here.
119#if defined(__x86_64__)
120 asm volatile(
121 "int $0x80"
122 : "=a"(ret)
123 : "a"(num), "b"(a0), "c"(a1), "d"(a2), "S"(a3)
124 : "memory", "cc"
125 );
126#elif defined(__aarch64__)
127 register long x8 asm("x8") = num;
128 register long x0 asm("x0") = a0;
129 register long x1 asm("x1") = a1;
130 register long x2 asm("x2") = a2;
131 register long x3 asm("x3") = a3;
132 register long x4 asm("x4") = 0;
133 asm volatile(
134 "svc #0"
135 : "+r"(x0)
136 : "r"(x8), "r"(x1), "r"(x2), "r"(x3), "r"(x4)
137 : "memory"
138 );
139 ret = x0;
140#elif defined(__riscv) && __riscv_xlen == 64
141 register long a7 asm("a7") = num;
142 register long a0_reg asm("a0") = a0;
143 register long a1_reg asm("a1") = a1;
144 register long a2_reg asm("a2") = a2;
145 register long a3_reg asm("a3") = a3;
146 asm volatile(
147 "ecall"
148 : "+r"(a0_reg)
149 : "r"(a7), "r"(a1_reg), "r"(a2_reg), "r"(a3_reg)
150 : "memory"
151 );
152 ret = a0_reg;
153#else
154# error "Unsupported architecture for __syscall5"
155#endif
156 return ret;
157}
