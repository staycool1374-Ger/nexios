53enum class VmmMapSteps : uint64_t {
54 Validate = 1ULL << 0,
55 Commit = 1ULL << 1,
56 Rollback = 1ULL << 2,
57 Reject = 1ULL << 3,
58 Done = 1ULL << 4,
59};

62inline constexpr uint64_t k_vmm_map_success_expected =
63 path_expected(VmmMapSteps::Validate, VmmMapSteps::Commit,
64 VmmMapSteps::Done);
65
68inline constexpr uint64_t k_vmm_map_rollback_expected =
69 path_expected(VmmMapSteps::Validate, VmmMapSteps::Rollback,
70 VmmMapSteps::Done);
71
73inline constexpr uint64_t k_vmm_map_reject_expected =
74 path_expected(VmmMapSteps::Reject, VmmMapSteps::Done);
75
76// Non-zero and pairwise-distinct (a fold typo would collapse these).
77static_assert(k_vmm_map_success_expected != 0);
78static_assert(k_vmm_map_success_expected != k_vmm_map_rollback_expected);
79static_assert(k_vmm_map_success_expected != k_vmm_map_reject_expected);
80static_assert(k_vmm_map_rollback_expected != k_vmm_map_reject_expected);
81
82// Bit-distinctness: bits 0..4 — a duplicated bit would hide a skipped step.
83static_assert((static_cast<uint64_t>(to_underlying(VmmMapSteps::Validate)) |
84 static_cast<uint64_t>(to_underlying(VmmMapSteps::Commit)) |
85 static_cast<uint64_t>(to_underlying(VmmMapSteps::Rollback)) |
86 static_cast<uint64_t>(to_underlying(VmmMapSteps::Reject)) |
87 static_cast<uint64_t>(to_underlying(VmmMapSteps::Done))) ==
88 0x1FULL);
