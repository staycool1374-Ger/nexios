36inline void cache_clean(const void *addr, size_t len) {
37 if (!addr || !len)
38 return;
39#if defined(CONFIG_ARCH_AARCH64)
40 uint64_t ctr = 0;
41 asm volatile("mrs %0, ctr_el0" : "=r"(ctr));
42 uint64_t line = 4ULL << ((ctr >> 16) & 0xFU);
43 uint64_t start = reinterpret_cast<uint64_t>(addr) & ~(line - 1);
44 uint64_t end = reinterpret_cast<uint64_t>(addr) + len;
45 for (uint64_t p = start; p < end; p += line) {
46 asm volatile("dc cvac, %0" : : "r"(p) : "memory");
47 }
48 asm volatile("dsb sy" : : : "memory");
49 asm volatile("isb");
50#elif defined(CONFIG_ARCH_RISCV64)
51#if defined(__riscv_zicbom)
52 // Block size comes from the DTB (riscv,cbom-block-size); the kernel
53 // build does not enable Zicbom today, so this arm is forward-proofing.
54 uint64_t start = reinterpret_cast<uint64_t>(addr);
55 uint64_t end = start + len;
56 for (uint64_t p = start; p < end; p += 64) {
57 asm volatile("cbo.clean 0(%0)" : : "r"(p) : "memory");
58 }
59 asm volatile("fence iorw, iorw" : : : "memory");
60#else
61 // No Zicbom in the build (-march=rv64imafdc): no cache-maintenance
62 // instruction exists — fence orders the (coherent-under-QEMU) accesses.
63 // Non-coherent silicon stays a #307 residual, documented in dma.hpp.
64 (void)addr;
65 (void)len;
66 asm volatile("fence iorw, iorw" : : : "memory");
67#endif
68#else
69 // x86_64: clflush per 64 B line (de-facto line size; stepping 64 hits
70 // every line of any power-of-2 size <= 64 at least once), mfence after.
71 uint64_t start = reinterpret_cast<uint64_t>(addr) & ~63ULL;
72 uint64_t end = reinterpret_cast<uint64_t>(addr) + len;
73 for (uint64_t p = start; p < end; p += 64) {
74 asm volatile("clflush (%0)" : : "r"(p) : "memory");
75 }
76 asm volatile("mfence" : : : "memory");
77#endif
78}
