58// Backing store: one chain per CPU (defined in path_tracer.cpp).

62[[nodiscard]] inline uint64_t& path_chain_slot() noexcept {
63 uint32_t cpu = static_cast<uint32_t>(arch::cpu_index());
64 if (cpu >= static_cast<uint32_t>(CONFIG_MAX_CPUS)) {
65 cpu = 0;
66 }
67 return g_path_chain[cpu];
68}
