66inline void yield_to_task(TaskControlBlock &task) noexcept {
67 arch::IrqGuard guard;
68 auto *original = Scheduler::current_task();
69 Scheduler::set_current(task);
70 __atomic_store_n(&scheduler_need_resched, true, __ATOMIC_RELEASE);
71 Scheduler::set_current(*original);
72 // set_current(task) re-enqueued `original` (the harness) into the ready
73 // queue because it was RUNNING at the time. We restored it as current
74 // again, so it must NOT remain in the ready queue (INV-4: a current task
75 // must never be queued, or next_task() re-selects it and the subsequent
76 // switch_to_task double-enqueue is refused, wedging the scheduler).
77 if (original->in_ready_queue_)
78 Scheduler::dequeue_ready(*original);
79 task.state = TaskState::READY;
80 Scheduler::enqueue_ready(task);
81}

194inline void wait_for_termination_safe(TaskControlBlock *task) {
195 if (!task)
196 return;
197 while (TaskControlBlock::is_valid(task) &&
198 task->state != TaskState::TERMINATED) {
199 __atomic_store_n(&kernel::scheduler_need_resched, true,
200 __ATOMIC_RELEASE);
201 arch::hlt();
202 }
203}

211inline void trigger_deadline_monitor_scan() {

213 // The monitor's block/wake handshake is unreliable under the
214 // deferred-switch + snapshot test environment: it can strand the monitor
215 // READY-but-not-in-runq after a wake, so the second scan in a test never
216 // completes and the harness spins forever. scan_deadlines() is the exact
217 // logic the monitor runs — calling it synchronously is deterministic.
