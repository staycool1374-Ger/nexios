66inline void yield_to_task(TaskControlBlock &task) noexcept {
67 // H2 stale-resume liveness guard (ROADMAP §v0.4.0 direction #4): the
68 // harness's stored context.rsp can occasionally point into a test body's
69 // setup path; on resume the harness re-executes this helper on a task that
70 // may already have self-terminated and been removed from id_table_. Acting
71 // on a dead task here corrupts state BEFORE any enqueue could be refused:
72 // set_current(dead) re-points the scheduler's current-cache, and
73 // `task.state = READY` resurrects a freed/recycled block. Refuse at entry
74 // so the stale resume is a harmless no-op (the test body's wait loop then
75 // exits because both tasks are TERMINATED / freed).
76 if (Scheduler::find_task(task.id) != &task)
77 return;
78
79 arch::IrqGuard guard;
80 auto *original = Scheduler::current_task();
81 Scheduler::set_current(task);
82 __atomic_store_n(&scheduler_need_resched, true, __ATOMIC_RELEASE);
83 Scheduler::set_current(*original);
84 // set_current(task) re-enqueued `original` (the harness) into the ready
85 // queue because it was RUNNING at the time. We restored it as current
86 // again, so it must NOT remain in the ready queue (INV-4: a current task
87 // must never be queued, or next_task() re-selects it and the subsequent
88 // switch_to_task double-enqueue is refused, wedging the scheduler).
89 if (original->in_ready_queue_)
90 Scheduler::dequeue_ready(*original);
91 task.state = TaskState::READY;
92 Scheduler::enqueue_ready(task);
93}

105class [[nodiscard]] ScopedCurrentTask {

109 public:

111 : saved_(Scheduler::current_task()) {

206inline void wait_for_termination_safe(TaskControlBlock *task) {
207 if (!task)
208 return;
209 while (TaskControlBlock::is_valid(task) &&
210 task->state != TaskState::TERMINATED) {
211 __atomic_store_n(&kernel::scheduler_need_resched, true,
212 __ATOMIC_RELEASE);
213 arch::hlt();
214 }
