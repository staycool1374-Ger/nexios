109constexpr bool validate_header(const ELF64Header *hdr) {
110 if (!hdr)
111 return false;
112 // Little-endian u32 assembly of ident[0..3] (all supported arches are
113 // LE; identical to the old reinterpret_cast read, constexpr-clean).
114 const uint32_t magic = static_cast<uint32_t>(hdr->ident[0]) |
115 (static_cast<uint32_t>(hdr->ident[1]) << 8) |
116 (static_cast<uint32_t>(hdr->ident[2]) << 16) |
117 (static_cast<uint32_t>(hdr->ident[3]) << 24);
118 if (magic != static_cast<uint32_t>(ELF_MAGIC))
119 return false;
120 if (hdr->ident[4] != 2)
121 return false; // 64-bit
122 if (hdr->ident[5] != 1)
123 return false; // little-endian
124 if (hdr->type != ET_EXEC && hdr->type != ET_DYN)
125 return false;
126#if defined(CONFIG_ARCH_AARCH64)
127 if (hdr->machine != 0xB7)
128 return false; // AArch64
129#elif defined(CONFIG_ARCH_RISCV64)
130 if (hdr->machine != 0xF3)
131 return false; // RISC-V (EM_RISCV)
132#else
133 if (hdr->machine != 0x3E)
134 return false; // x86_64 (default)
135#endif
136 if (hdr->version == 0)
137 return false;
138 if (hdr->phnum > 64)
139 return false;
140 if (hdr->ehsize < sizeof(ELF64Header))
141 return false;
142 if (hdr->phentsize < sizeof(ELF64ProgramHeader))
143 return false;
144 if (hdr->phnum > 0) {
145 // Check for arithmetic overflow in program header region end
146 uint64_t ph_end =
147 hdr->phoff + static_cast<uint64_t>(hdr->phnum) * hdr->phentsize;
148 if (ph_end < hdr->phoff)
149 return false; // overflow
150 // Ensure header fits within reasonable image size
151 if (ph_end > 1_MiB)
152 return false;
153 }
154 // Verify ident padding bytes are zero
155 for (int i = 7; i < 16; ++i) {
156 if (hdr->ident[i] != 0)
157 return false;
158 }
159 // Reject files with suspicious entry point
160 if (hdr->entry >= USER_SPACE_LIMIT && hdr->entry != 0)
161 return false;
162 return true;
163}

167constexpr bool validate_segment(const ELF64ProgramHeader *phdr, uint64_t file_size) {
168 if (phdr->type != PT_LOAD)
169 return true;
170 // Basic size sanity
171 if (phdr->filesz > phdr->memsz)
172 return false;
173 // Overflow checks
174 if (phdr->offset + phdr->filesz < phdr->offset)
175 return false;
176 if (phdr->vaddr + phdr->memsz < phdr->vaddr)
177 return false;
178 if (phdr->vaddr + phdr->filesz < phdr->vaddr)
179 return false;
180 // VULN-H2: the segment's file-resident bytes must fit inside the actual
181 // file buffer. Previously only a 4_MiB constant bounded phdr->offset;
182 // a crafted ELF could read past the PMM::alloc_contiguous() buffer.
183 // BSS-only segments (filesz == 0) carry no bytes: a 64 KiB-aligned
184 // linker may place their offset past EOF (issue #356) without any
185 // read occurring, so the bound applies only when bytes are consumed.
186 if (file_size != 0 && phdr->filesz != 0 &&
187 phdr->offset + phdr->filesz > file_size)
188 return false;
189 // Must be within user space
190 if (phdr->vaddr >= USER_SPACE_LIMIT)
191 return false;
192 if (phdr->vaddr + phdr->memsz > USER_SPACE_LIMIT)
193 return false;
194 // Reject zero-size segments in memory
195 if (phdr->memsz == 0)
196 return false;
197 // Reject segments that are too large (> 64 MiB)
198 if (phdr->memsz > 64_MiB)
199 return false;
200 // Reject segments with suspicious alignment
201 if (phdr->align > 1_MiB)
202 return false;
203 // Validate file offset is within the image (cannot check exact size
204 // without file length,
205 // but ensure it's reasonable: less than a typical max ELF size)
206 if (phdr->offset > 4_MiB)
207 return false;
208 // W^X enforcement: segment should not be both writable AND executable
209 if ((phdr->flags & 2) && (phdr->flags & 1))
210 return false;
211 return true;
212}
