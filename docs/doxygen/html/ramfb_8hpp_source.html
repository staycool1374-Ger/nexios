37struct RamfbConfig {
38 uint64_t addr_be;
39 uint32_t fourcc_be;
40 uint32_t flags_be;
41 uint32_t width_be;
42 uint32_t height_be;
43 uint32_t stride_be;

53constexpr RamfbConfig make_ramfb_config(uint64_t phys, uint32_t width,
54 uint32_t height) {
55 RamfbConfig cfg{};
56 cfg.addr_be = __builtin_bswap64(phys);
57 cfg.fourcc_be = __builtin_bswap32(RAMFB_FOURCC_XRGB8888);
58 cfg.flags_be = 0;
59 cfg.width_be = __builtin_bswap32(width);
60 cfg.height_be = __builtin_bswap32(height);
61 cfg.stride_be = __builtin_bswap32(width * 4);
62 return cfg;
63}
