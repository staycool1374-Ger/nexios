39// Wire the window tables into the boot kernel PML4/TTBR1 (lazy, once).
40// Returns true when the window is usable; false means unusable on this
41// arch/build (caller takes the exhaustion-0 path — no arch guard outside).
42// P1-C1: all-or-nothing re-entry — partial wiring completes on retry,
43// usable is reported only when fully wired; failure is loud-once
44// (one-time serial/boot diagnostic), never silent.
45// Must be idempotent: a second call with tables present re-adopts them.
46bool wire_window_tables();
47
48// Map/unmap one 4 KiB kernel-stack page at @p virt.
49// Out-of-window VAs are a silent no-op (matches x86 pt_idx>=8 rule).
50void map_window_page(uint64_t virt, uint64_t phys);
51void unmap_window_page(uint64_t virt);
52
53// TLB maintenance for one window VA (invlpg / tlbi vae1 / sfence.vma).
54void flush_window_va(uint64_t virt);
55
56// Snapshot support: byte size, capture, restore (restore flushes each VA
57// via flush_window_va). Layout is backend-private; task.cpp treats it as
58// opaque bytes appended after its own pool/bump/lists framing.
59size_t snapshot_bytes();
60void snapshot_capture(uint8_t *dst);
61void snapshot_restore(const uint8_t *src);
