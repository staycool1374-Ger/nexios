43class ArchInterruptController {
44 public:
46 static inline void init() {
47 outb(arch::PIC1_CMD, 0x11);
48 outb(arch::PIC2_CMD, 0x11);
49 outb(arch::PIC1_DATA, 0x20);
50 outb(arch::PIC2_DATA, 0x28);
51 outb(arch::PIC1_DATA, 0x04);
52 outb(arch::PIC2_DATA, 0x02);
53 outb(arch::PIC1_DATA, 0x01);
54 outb(arch::PIC2_DATA, 0x01);
55 outb(arch::PIC1_DATA, 0x00);
56 outb(arch::PIC2_DATA, 0x00);
57 }
58
61 static inline void eoi(uint8_t vector) {
62 outb(arch::PIC1_CMD, 0x20);
63 if (vector >= 40)
64 outb(arch::PIC2_CMD, 0x20);
65 }
66
69 static inline void mask(uint8_t irq) {
70 uint16_t port = (irq < 8) ? arch::PIC1_DATA : arch::PIC2_DATA;
71 uint8_t bit = 1 << (irq & 7);
72 uint8_t m = inb(port) | bit;
73 outb(port, m);
74 }
75
78 static inline void unmask(uint8_t irq) {
79 uint16_t port = (irq < 8) ? arch::PIC1_DATA : arch::PIC2_DATA;
80 uint8_t bit = ~(1 << (irq & 7));
81 uint8_t m = inb(port) & bit;
82 outb(port, m);
83 }
84
87 static inline IrqState snapshot() {
88 IrqState s;
89 s.pic1_mask = inb(arch::PIC1_DATA);
90 s.pic2_mask = inb(arch::PIC2_DATA);
91 return s;
92 }
93
96 static inline uint16_t max_vector() {
97 return 47;
98 }
99
102 static inline bool line_enabled(uint8_t irq) {
103 if (irq >= 16)
104 return false;
105 uint16_t port = (irq < 8) ? arch::PIC1_DATA : arch::PIC2_DATA;
106 return (inb(port) & static_cast<uint8_t>(1u << (irq & 7u))) == 0;
107 }
108
111 static inline void restore(const IrqState &state) {
112 outb(arch::PIC1_DATA, state.pic1_mask);
113 outb(arch::PIC2_DATA, state.pic2_mask);
114 }
115};
