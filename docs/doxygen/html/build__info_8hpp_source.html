44constexpr bool streq(const char *left, const char *right) {
45 for (int idx = 0; idx < 16; ++idx) {
46 if (left[idx] != right[idx])
47 return false;
48 if (left[idx] == '\0')
49 return true;
50 }
51 return left[16] == '\0' && right[16] == '\0';
52}
