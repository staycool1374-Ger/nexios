45inline uint64_t pci_ecam_addr(PciBdf bdf, uint8_t reg) {
46 return PCI_ECAM_BASE | (static_cast<uint64_t>(bdf.bus) << 20) |
47 (static_cast<uint64_t>(bdf.device) << 15) |
48 (static_cast<uint64_t>(bdf.function) << 12) | reg;
49}
