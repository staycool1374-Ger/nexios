54inline bool send_resched_ipi(uint64_t cpu) noexcept {
55#if defined(CONFIG_ARCH_X86_64)
56 if (cpu >= CONFIG_MAX_CPUS) {
57 return false;
58 }
59 const uint32_t lapic = (cpu == cpu_index()) ? APIC::lapic_id()
60 : per_cpu[cpu].lapic_id;
61 return APIC::send_ipi(lapic, APIC::SCHED_VECTOR, APIC::IpiMode::FIXED);
62#elif defined(CONFIG_ARCH_AARCH64)
63 if (cpu >= CONFIG_MAX_CPUS) {
64 return false;
65 }
66 const uint32_t target = static_cast<uint32_t>(
67 kernel::smp::ap_aff0(static_cast<uint8_t>(cpu)) & 0xFFU);
68 return gic_send_sgi(target, SMP_SCHED_SGI);
69#elif defined(CONFIG_ARCH_RISCV64)
70 return kernel::smp::send_resched_ipi(cpu);
71#else
72 (void)cpu;
73 return false;
74#endif
75}
