234inline const DmesgRecord *catalog_lookup(ErrorSubsystem subsys,
235 uint64_t error_nr) noexcept { const DmesgRecord *table = nullptr;
236 size_t count = 0;
237 switch (subsys) {
238 case ErrorSubsystem::DAEMON:
239 table = kDmesgCatalog_DAEMON;
240 count = sizeof(kDmesgCatalog_DAEMON) / sizeof(kDmesgCatalog_DAEMON[0]);
241 break;
242 case ErrorSubsystem::ELF:
243 table = kDmesgCatalog_ELF;
244 count = sizeof(kDmesgCatalog_ELF) / sizeof(kDmesgCatalog_ELF[0]);
245 break;
246 case ErrorSubsystem::USER:
247 table = kDmesgCatalog_USER;
248 count = sizeof(kDmesgCatalog_USER) / sizeof(kDmesgCatalog_USER[0]);
249 break;
250 case ErrorSubsystem::NET:
251 table = kDmesgCatalog_NET;
252 count = sizeof(kDmesgCatalog_NET) / sizeof(kDmesgCatalog_NET[0]);
253 break;
254 case ErrorSubsystem::DRIVER:
255 table = kDmesgCatalog_DRIVER;
256 count =
257 sizeof(kDmesgCatalog_DRIVER) / sizeof(kDmesgCatalog_DRIVER[0]);
258 break;
259 case ErrorSubsystem::INIT:
260 table = kDmesgCatalog_INIT;
261 count = sizeof(kDmesgCatalog_INIT) / sizeof(kDmesgCatalog_INIT[0]);
262 break;
263 case ErrorSubsystem::TIMING:
264 table = kDmesgCatalog_TIMING;
265 count =
266 sizeof(kDmesgCatalog_TIMING) / sizeof(kDmesgCatalog_TIMING[0]);
267 break;
268 case ErrorSubsystem::TEST:
269 table = kDmesgCatalog_TEST;
270 count = sizeof(kDmesgCatalog_TEST) / sizeof(kDmesgCatalog_TEST[0]);
271 break;
272 case ErrorSubsystem::PMM:
273 table = kDmesgCatalog_PMM;
274 count = sizeof(kDmesgCatalog_PMM) / sizeof(kDmesgCatalog_PMM[0]);
275 break;
276 default:
277 return nullptr;
278 }
279 for (size_t idx = 0; idx < count; ++idx) {
280 if (table[idx].error_nr == error_nr) {
281 return &table[idx];
282 }
283 }
284 return nullptr;
285}

291inline const char *catalog_text(ErrorSubsystem subsys,
292 uint64_t error_nr) noexcept {
293 const DmesgRecord *found = catalog_lookup(subsys, error_nr);
294 return found ? found->text : "UNKNOWN";

303inline LogSeverity lookup_severity(ErrorSubsystem subsys,
304 uint64_t code) noexcept {
305 const DmesgRecord *found = catalog_lookup(subsys, code);
306 if (found) {
307 return found->severity;
308 }
309 if (subsys == ErrorSubsystem::BASE) {
310 return base_code_is_info(code) ? LogSeverity::INFO : LogSeverity::ERROR;
311 }
312 const uint64_t raw = strip_canonical(subsys, code);
313 if (raw == 0) {
314 return LogSeverity::INFO;
315 }
316 switch (subsys) {
317 case ErrorSubsystem::SYNC:
318 switch (raw) {
319 case 3:
320 case 7:
321 case 9:
322 case 12:
323 case 13:
324 case 14:
325 return LogSeverity::WARN;
326 default:
327 return LogSeverity::ERROR;
328 }
329 case ErrorSubsystem::VFS:
330 switch (raw) {
331 case 3:
332 case 4:
333 case 5:
334 case 6:
335 case 7:
336 return LogSeverity::WARN;
337 default:
338 return LogSeverity::ERROR;
339 }
340 case ErrorSubsystem::SCHED:
341 return raw == 12 ? LogSeverity::WARN : LogSeverity::ERROR;
342 case ErrorSubsystem::IPC:
343 return raw == 3 ? LogSeverity::WARN : LogSeverity::ERROR;
344 case ErrorSubsystem::SYSCALL:
345 switch (raw) {
346 case 403:
347 case 407:
348 case 409:
349 case 412:
350 case 413:
351 case 414:
352 case 503:
353 case 504:
354 case 505:
355 case 506:
356 case 507:
357 case 603:
358 return LogSeverity::WARN;
359 default:
360 return LogSeverity::ERROR;
361 }
362 default:
363 return LogSeverity::ERROR;
364 }
365}

369template <size_t Count>
370constexpr bool catalog_table_valid(const DmesgRecord (&table)[Count]) {
371 for (size_t idx = 0; idx < Count; ++idx) {
372 if (table[idx].error_nr < 7000 || table[idx].error_nr >= 19000) {
373 return false;
374 }
375 for (size_t inner = idx + 1; inner < Count; ++inner) {
376 if (table[idx].subsystem == table[inner].subsystem &&
377 table[idx].error_nr == table[inner].error_nr) {
378 return false;
379 }
380 }
381 }
382 return true;
383}
