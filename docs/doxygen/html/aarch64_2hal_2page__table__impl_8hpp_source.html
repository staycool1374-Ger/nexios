175inline uint64_t *ArchPageTable::get_table(uint64_t table_base, uint64_t index,
176 bool create) {
177 // Page-table pages are physical; access them through the HHDM direct-map
178 // alias so the walk is independent of the currently-active TTBR0 (which
179 // may be a user/private PML4 without the identity map).
180 uint64_t *table = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
181 table_base);
182 uint64_t entry = table[index];
183
184 if (entry & DESC_VALID) {
185 if (entry & DESC_TABLE) {
186 return reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
187 (entry & DESC_PHYS_MASK));
188 }
189 return nullptr;
190 }
191
192 if (!create)
193 return nullptr;
194
195 uint64_t new_page = kernel::PMM::alloc_page_table();
196 if (!new_page)
197 return nullptr;
198
199 void *new_page_ptr =
200 reinterpret_cast<void *>(arch::HHDM_OFFSET + new_page);
201 memset(new_page_ptr, 0, PAGE_SIZE);
202 table[index] = new_page | DESC_VALID | DESC_TABLE;
203 return reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + new_page);
204}

206inline uint64_t ArchPageTable::attr_from_flags(uint64_t flags) {
207 uint64_t attr = DESC_VALID | DESC_AF | DESC_SH_INNER | DESC_NS;
208 if (flags & kernel::PageFlags::PRESENT)
209 attr |= DESC_VALID;
210 if (flags & kernel::PageFlags::WRITE)
211 attr |= AP_RW;
212 else
213 attr |= AP_RO;
214 if (flags & kernel::PageFlags::USER) {
215 // User accessible; PXN (SMEP parity — EL1 must not execute user
216 // pages). UXN left clear so EL0 may execute.
217 attr |= PXN;
218 } else {
219 attr |= UXN | PXN;
220 }
221 if (flags & kernel::PageFlags::NX)
222 attr |= UXN | PXN;
223 if (flags & kernel::PageFlags::CACHE_DISABLED)
224 attr |= ATTR_IDX_DEVICE;
225 else
226 attr |= ATTR_IDX_NORMAL;
227 return attr;
228}

250inline kernel::ErrorOr<void>
251ArchPageTable::map_page(uint64_t virt, uint64_t phys, uint64_t flags) {
252 uint64_t l0_idx = (virt >> L0_SHIFT) & TABLE_MASK;
253 uint64_t l1_idx = (virt >> L1_SHIFT) & TABLE_MASK;
254 uint64_t l2_idx = (virt >> L2_SHIFT) & TABLE_MASK;
255 uint64_t l3_idx = (virt >> L3_SHIFT) & TABLE_MASK;
256
257 uint64_t l0_phys = current();
258 uint64_t *l0 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l0_phys);
259
260 uint64_t *l1 = walk_l0(l0_phys, l0_idx, true);
261 if (!l1)
262 return kernel::Error::OOM;
263 uint64_t l1_phys = l0[l0_idx] & DESC_PHYS_MASK;
264
265 uint64_t *l2 = walk_l1(l1_phys, l1_idx, true);
266 if (!l2)
267 return kernel::Error::OOM;
268 uint64_t l2_phys = l1[l1_idx] & DESC_PHYS_MASK;
269
270 uint64_t entry_flags = attr_from_flags(flags);
271
272 if ((virt & ((1ULL << L2_SHIFT) - 1)) == 0 &&
273 (phys & ((1ULL << L2_SHIFT) - 1)) == 0) {
274 l2[l2_idx] = phys | entry_flags | DESC_BLOCK | (1ULL << 10);
275 dsb_sy();
276 isb();
277 return {};
278 }
279
280 uint64_t *l3 = walk_l2(l2_phys, l2_idx, true);
281 if (!l3)
282 return kernel::Error::OOM;
283
284 l3[l3_idx] = phys | entry_flags | DESC_TABLE | (1ULL << 10);
285 dsb_sy();
286 isb();
287 return {};
288}

290inline kernel::ErrorOr<void> ArchPageTable::unmap_page(uint64_t virt) {
291 uint64_t l0_idx = (virt >> L0_SHIFT) & TABLE_MASK;
292 uint64_t l1_idx = (virt >> L1_SHIFT) & TABLE_MASK;
293 uint64_t l2_idx = (virt >> L2_SHIFT) & TABLE_MASK;
294 uint64_t l3_idx = (virt >> L3_SHIFT) & TABLE_MASK;
295
296 uint64_t l0_phys = current();
297 uint64_t *l0 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l0_phys);
298
299 if (!(l0[l0_idx] & DESC_VALID))
300 return kernel::Error::NOT_FOUND;
301 uint64_t *l1 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
302 (l0[l0_idx] & DESC_PHYS_MASK));
303 if (!(l1[l1_idx] & DESC_VALID))
304 return kernel::Error::NOT_FOUND;
305 uint64_t l2_phys = l1[l1_idx] & DESC_PHYS_MASK;
306 uint64_t *l2 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l2_phys);
307
308 if ((virt & ((1ULL << L2_SHIFT) - 1)) == 0 && (l2[l2_idx] & DESC_BLOCK)) {
309 l2[l2_idx] = 0;
310 dsb_sy();
311 isb();
312 return {};
313 }
314
315 if (!(l2[l2_idx] & DESC_VALID))
316 return kernel::Error::NOT_FOUND;
317 uint64_t l3_phys = l2[l2_idx] & DESC_PHYS_MASK;
318 uint64_t *l3 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l3_phys);
319
320 if (!(l3[l3_idx] & DESC_VALID))
321 return kernel::Error::NOT_FOUND;
322 l3[l3_idx] = 0;
323 dsb_sy();
324 isb();
325 // The leaf translation for `virt` is gone; invalidate it so no stale TLB
326 // entry can keep resolving into a freed/reused table page.
327 arch_page_table_tlb_flush(virt);
328
329 // Reclaim intermediate tables once they become entirely empty. Only
330 // map-allocated tables reach the empty state (the boot L1/L2 pages stay
331 // populated), so freeing them here never releases a boot-static page.
332 // Guard on DESC_TABLE: a 2 MiB block descriptor (bit[1]=0) at L2 must
333 // never be misinterpreted as an L3 table pointer and freed.
334 if ((l2[l2_idx] & DESC_TABLE) != 0) {
335 bool l3_empty = true;
336 for (size_t i = 0; i < ENTRIES; ++i) {
337 if (l3[i] != 0) {
338 l3_empty = false;
339 break;
340 }
341 }
342 if (l3_empty) {
343 l2[l2_idx] = 0;
344 kernel::PMM::free_page(l3_phys);
345 bool l2_empty = true;
346 for (size_t i = 0; i < ENTRIES; ++i) {
347 if (l2[i] != 0) {
348 l2_empty = false;
349 break;
350 }
351 }
352 if (l2_empty) {
353 l1[l1_idx] = 0;
354 kernel::PMM::free_page(l2_phys);
355 }
356 }
357 }
358 dsb_sy();
359 isb();
360 return {};
361}

363inline uint64_t ArchPageTable::get_physical(uint64_t virt) {
364 uint64_t l0_idx = (virt >> L0_SHIFT) & TABLE_MASK;
365 uint64_t l1_idx = (virt >> L1_SHIFT) & TABLE_MASK;
366 uint64_t l2_idx = (virt >> L2_SHIFT) & TABLE_MASK;
367 uint64_t l3_idx = (virt >> L3_SHIFT) & TABLE_MASK;
368
369 uint64_t l0_phys = current();
370 uint64_t *l0 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l0_phys);
371
372 if (!(l0[l0_idx] & DESC_VALID))
373 return 0;
374 uint64_t *l1 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
375 (l0[l0_idx] & DESC_PHYS_MASK));
376 if (!(l1[l1_idx] & DESC_VALID))
377 return 0;
378 uint64_t *l2 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
379 (l1[l1_idx] & DESC_PHYS_MASK));
380
381 if ((virt & ((1ULL << L2_SHIFT) - 1)) == 0 && (l2[l2_idx] & DESC_BLOCK)) {
382 return (l2[l2_idx] & DESC_BLOCK_MASK) | (virt & 0x1FFFFF);
383 }
384
385 if (!(l2[l2_idx] & DESC_VALID))
386 return 0;
387 uint64_t *l3 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
388 (l2[l2_idx] & DESC_PHYS_MASK));
389 if (!(l3[l3_idx] & DESC_VALID))
390 return 0;
391
392 return (l3[l3_idx] & DESC_PHYS_MASK) | (virt & 0xFFF);
393}
