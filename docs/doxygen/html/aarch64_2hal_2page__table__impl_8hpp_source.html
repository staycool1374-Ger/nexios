220inline uint64_t *ArchPageTable::get_table(uint64_t table_base, uint64_t index,
221 bool create) {
222 // Page-table pages are physical; access them through the HHDM direct-map
223 // alias so the walk is independent of the currently-active TTBR0 (which
224 // may be a user/private PML4 without the identity map).
225 uint64_t *table = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
226 table_base);
227 uint64_t entry = table[index];
228
229 if (entry & DESC_VALID) {
230 if (entry & DESC_TABLE) {
231 return reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
232 (entry & DESC_PHYS_MASK));
233 }
234 return nullptr;
235 }
236
237 if (!create)
238 return nullptr;
239
240 uint64_t new_page = kernel::PMM::alloc_page_table();
241 if (!new_page)
242 return nullptr;
243
244 void *new_page_ptr =
245 reinterpret_cast<void *>(arch::HHDM_OFFSET + new_page);
246 memset(new_page_ptr, 0, PAGE_SIZE);
247 table[index] = new_page | DESC_VALID | DESC_TABLE;
248 return reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + new_page);
249}

251inline uint64_t ArchPageTable::attr_from_flags(uint64_t flags) {
252 uint64_t attr = DESC_VALID | DESC_AF | DESC_SH_INNER | DESC_NS;
253 if (flags & kernel::PageFlags::PRESENT)
254 attr |= DESC_VALID;
255 if (flags & kernel::PageFlags::WRITE)
256 attr |= AP_RW;
257 else
258 attr |= AP_RO;
259 if (flags & kernel::PageFlags::USER) {
260 // User accessible; PXN (SMEP parity — EL1 must not execute user
261 // pages). UXN left clear so EL0 may execute.
262 attr |= PXN;
263 } else {
264 attr |= UXN | PXN;
265 }
266 if (flags & kernel::PageFlags::NX)
267 attr |= UXN | PXN;
268 if (flags & kernel::PageFlags::CACHE_DISABLED)
269 attr |= ATTR_IDX_DEVICE;
270 else
271 attr |= ATTR_IDX_NORMAL;
272 return attr;
273}

295inline bool ArchPageTable::split_l1_block(uint64_t *l1, size_t idx) {
296 const uint64_t block = l1[idx];
297 const uint64_t base = block & DESC_L1_BLOCK_MASK;
298 const uint64_t preserved = block & ~DESC_L1_BLOCK_MASK;
299 uint64_t split_phys = kernel::PMM::alloc_page_table();
300 if (!split_phys)
301 return false;
302 auto *l2 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + split_phys);
303 for (size_t i = 0; i < 512; ++i)
304 l2[i] = (base + i * 0x200000ULL) | preserved | DESC_AF;
305 l1[idx] = split_phys | DESC_VALID | DESC_TABLE;
306 dsb_sy();
307 isb();
308 return true;
309}

311inline bool ArchPageTable::split_l2_block(uint64_t *l2, size_t idx) {
312 const uint64_t block = l2[idx];
313 const uint64_t base = block & DESC_BLOCK_MASK;
314 const uint64_t preserved = block & ~DESC_BLOCK_MASK;
315 uint64_t split_phys = kernel::PMM::alloc_page_table();
316 if (!split_phys)
317 return false;
318 auto *l3 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + split_phys);
319 for (size_t i = 0; i < 512; ++i)
320 l3[i] = (base + i * 0x1000ULL) | preserved | DESC_TABLE | DESC_AF;
321 l2[idx] = split_phys | DESC_VALID | DESC_TABLE;
322 dsb_sy();
323 isb();
324 return true;

327inline kernel::ErrorOr<void>
328ArchPageTable::map_page(uint64_t virt, uint64_t phys, uint64_t flags) {
329 uint64_t l0_idx = (virt >> L0_SHIFT) & TABLE_MASK;
330 uint64_t l1_idx = (virt >> L1_SHIFT) & TABLE_MASK;
331 uint64_t l2_idx = (virt >> L2_SHIFT) & TABLE_MASK;
332 uint64_t l3_idx = (virt >> L3_SHIFT) & TABLE_MASK;
333
334 uint64_t l0_phys = current();
335 uint64_t *l0 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l0_phys);
336
337 uint64_t *l1 = walk_l0(l0_phys, l0_idx, true);
338 if (!l1)
339 return kernel::Error::OOM;
340 // Issue #352: split a user-range L1 1 GiB block so deeper levels can
341 // be installed. Kernel-half blocks stay fail-closed (get_table in
342 // walk_l1 refuses them as before when this guard does not fire).
343 if (virt < USER_VA_LIMIT && (l1[l1_idx] & DESC_VALID) &&
344 ((l1[l1_idx] & DESC_TABLE) == 0)) {
345 if (!split_l1_block(l1, l1_idx))
346 return kernel::Error::OOM;
347 }
348 uint64_t l1_phys = l0[l0_idx] & DESC_PHYS_MASK;
349
350 uint64_t *l2 = walk_l1(l1_phys, l1_idx, true);
351 if (!l2)
352 return kernel::Error::OOM;
353 // Issue #352: split a user-range L2 2 MiB block the same way.
354 if (virt < USER_VA_LIMIT && (l2[l2_idx] & DESC_VALID) &&
355 ((l2[l2_idx] & DESC_TABLE) == 0)) {
356 if (!split_l2_block(l2, l2_idx))
357 return kernel::Error::OOM;
358 }
359 uint64_t l2_phys = l1[l1_idx] & DESC_PHYS_MASK;
360
361 uint64_t entry_flags = attr_from_flags(flags);
362
363 if ((virt & ((1ULL << L2_SHIFT) - 1)) == 0 &&
364 (phys & ((1ULL << L2_SHIFT) - 1)) == 0) {
365 l2[l2_idx] = phys | entry_flags | DESC_BLOCK | (1ULL << 10);
366 dsb_sy();
367 // Audit S3 cure (issue #352): installing over a live entry (e.g.
368 // after a block split, or a block-over-table remap) must retire
369 // any stale TLB entry for this VA — unmap flushes, map must too.
370 tlbi_vae1(virt);
371 dsb_sy();
372 isb();
373 return {};
374 }
375
376 uint64_t *l3 = walk_l2(l2_phys, l2_idx, true);
377 if (!l3)
378 return kernel::Error::OOM;
379
380 l3[l3_idx] = phys | entry_flags | DESC_TABLE | (1ULL << 10);
381 dsb_sy();
382 tlbi_vae1(virt);
383 dsb_sy();
384 isb();
385 return {};
386}

388inline kernel::ErrorOr<void> ArchPageTable::unmap_page(uint64_t virt) {
389 uint64_t l0_idx = (virt >> L0_SHIFT) & TABLE_MASK;
390 uint64_t l1_idx = (virt >> L1_SHIFT) & TABLE_MASK;
391 uint64_t l2_idx = (virt >> L2_SHIFT) & TABLE_MASK;
392 uint64_t l3_idx = (virt >> L3_SHIFT) & TABLE_MASK;
393
394 uint64_t l0_phys = current();
395 uint64_t *l0 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l0_phys);
396
397 if (!(l0[l0_idx] & DESC_VALID))
398 return kernel::Error::NOT_FOUND;
399 uint64_t *l1 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
400 (l0[l0_idx] & DESC_PHYS_MASK));
401 if (!(l1[l1_idx] & DESC_VALID))
402 return kernel::Error::NOT_FOUND;
403 uint64_t l2_phys = l1[l1_idx] & DESC_PHYS_MASK;
404 uint64_t *l2 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l2_phys);
405
406 if ((virt & ((1ULL << L2_SHIFT) - 1)) == 0 && (l2[l2_idx] & DESC_BLOCK)) {
407 l2[l2_idx] = 0;
408 dsb_sy();
409 isb();
410 return {};
411 }
412
413 if (!(l2[l2_idx] & DESC_VALID))
414 return kernel::Error::NOT_FOUND;
415 uint64_t l3_phys = l2[l2_idx] & DESC_PHYS_MASK;
416 uint64_t *l3 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l3_phys);
417
418 if (!(l3[l3_idx] & DESC_VALID))
419 return kernel::Error::NOT_FOUND;
420 l3[l3_idx] = 0;
421 dsb_sy();
422 isb();
423 // The leaf translation for `virt` is gone; invalidate it so no stale TLB
424 // entry can keep resolving into a freed/reused table page.
425 arch_page_table_tlb_flush(virt);
426
427 // Reclaim intermediate tables once they become entirely empty. Only
428 // map-allocated tables reach the empty state (the boot L1/L2 pages stay
429 // populated), so freeing them here never releases a boot-static page.
430 // Guard on DESC_TABLE: a 2 MiB block descriptor (bit[1]=0) at L2 must
431 // never be misinterpreted as an L3 table pointer and freed.
432 if ((l2[l2_idx] & DESC_TABLE) != 0) {
433 bool l3_empty = true;
434 for (size_t i = 0; i < ENTRIES; ++i) {
435 if (l3[i] != 0) {
436 l3_empty = false;
437 break;
438 }
439 }
440 if (l3_empty) {
441 l2[l2_idx] = 0;
442 kernel::PMM::free_page(l3_phys);
443 bool l2_empty = true;
444 for (size_t i = 0; i < ENTRIES; ++i) {
445 if (l2[i] != 0) {
446 l2_empty = false;
447 break;
448 }
449 }
450 if (l2_empty) {
451 l1[l1_idx] = 0;
452 kernel::PMM::free_page(l2_phys);
453 }
454 }
455 }
456 dsb_sy();
457 isb();
458 return {};
459}

461inline uint64_t ArchPageTable::get_physical(uint64_t virt) {
462 uint64_t l0_idx = (virt >> L0_SHIFT) & TABLE_MASK;
463 uint64_t l1_idx = (virt >> L1_SHIFT) & TABLE_MASK;
464 uint64_t l2_idx = (virt >> L2_SHIFT) & TABLE_MASK;
465 uint64_t l3_idx = (virt >> L3_SHIFT) & TABLE_MASK;
466
467 uint64_t l0_phys = current();
468 uint64_t *l0 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET + l0_phys);
469
470 if (!(l0[l0_idx] & DESC_VALID))
471 return 0;
472 uint64_t *l1 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
473 (l0[l0_idx] & DESC_PHYS_MASK));
474 if (!(l1[l1_idx] & DESC_VALID))
475 return 0;
476 uint64_t *l2 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
477 (l1[l1_idx] & DESC_PHYS_MASK));
478
479 if ((virt & ((1ULL << L2_SHIFT) - 1)) == 0 && (l2[l2_idx] & DESC_BLOCK)) {
480 return (l2[l2_idx] & DESC_BLOCK_MASK) | (virt & 0x1FFFFF);
481 }
482
483 if (!(l2[l2_idx] & DESC_VALID))
484 return 0;
485 uint64_t *l3 = reinterpret_cast<uint64_t *>(arch::HHDM_OFFSET +
486 (l2[l2_idx] & DESC_PHYS_MASK));
487 if (!(l3[l3_idx] & DESC_VALID))
488 return 0;
489
490 return (l3[l3_idx] & DESC_PHYS_MASK) | (virt & 0xFFF);
491}
