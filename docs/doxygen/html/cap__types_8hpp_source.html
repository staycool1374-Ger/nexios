58constexpr uint32_t cap_slot_bits() noexcept {
59 uint32_t bits = 0;
60 uint32_t slots = static_cast<uint32_t>(CONFIG_CSLOT_COUNT);
61 while ((1u << bits) < slots)
62 ++bits;
63 return bits;
64}

66constexpr uint32_t CAP_SLOT_BITS = cap_slot_bits();
67constexpr uint32_t CAP_SLOT_MASK = (1u << CAP_SLOT_BITS) - 1u;

81constexpr uint64_t encode_handle(uint32_t cspace_id, uint32_t slot_index,
82 uint32_t gen) noexcept {
83 constexpr uint32_t shift = CAP_HANDLE_IDBITS + CAP_SLOT_BITS;
84 return (static_cast<uint64_t>(gen) << shift) |
85 (static_cast<uint64_t>(cspace_id & 0xFFu) << CAP_SLOT_BITS) |
86 (static_cast<uint64_t>(slot_index) & CAP_SLOT_MASK);
87}

90constexpr uint32_t handle_slot(uint64_t handle) noexcept {

95constexpr uint32_t handle_cspace(uint64_t handle) noexcept {

100constexpr uint32_t handle_gen(uint64_t handle) noexcept {
