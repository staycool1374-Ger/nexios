94constexpr uint32_t cap_slot_bits() noexcept {
95 uint32_t bits = 0;
96 uint32_t slots = static_cast<uint32_t>(CONFIG_CSLOT_COUNT);
97 while ((1u << bits) < slots)
98 ++bits;
99 return bits;
100}

102constexpr uint32_t CAP_SLOT_BITS = cap_slot_bits();
103constexpr uint32_t CAP_SLOT_MASK = (1u << CAP_SLOT_BITS) - 1u;

117constexpr uint64_t encode_handle(uint32_t cspace_id, uint32_t slot_index,
118 uint32_t gen) noexcept {
119 constexpr uint32_t shift = CAP_HANDLE_IDBITS + CAP_SLOT_BITS;
120 return (static_cast<uint64_t>(gen) << shift) |
121 (static_cast<uint64_t>(cspace_id & 0xFFu) << CAP_SLOT_BITS) |
122 (static_cast<uint64_t>(slot_index) & CAP_SLOT_MASK);
123}

126constexpr uint32_t handle_slot(uint64_t handle) noexcept {
127 return static_cast<uint32_t>(handle & CAP_SLOT_MASK);
128}

131constexpr uint32_t handle_cspace(uint64_t handle) noexcept {
132 return static_cast<uint32_t>((handle >> CAP_SLOT_BITS) & 0xFFu);
133}

136constexpr uint32_t handle_gen(uint64_t handle) noexcept {
137 return static_cast<uint32_t>(handle >> (CAP_HANDLE_IDBITS + CAP_SLOT_BITS));
138}
