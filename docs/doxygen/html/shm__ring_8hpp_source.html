38constexpr uint64_t SHM_RING_MAGIC = 0x4E455852494E47ULL;
40constexpr uint64_t SHM_RING_VERSION = 1;

46struct alignas(64) SharedRingHeader {
48 uint64_t magic = 0;
50 uint64_t version = 0;
53 uint64_t capacity = 0;
55 uint64_t element_size = 0;
57 alignas(64) uint64_t head = 0;
59 alignas(64) uint64_t tail = 0;
60};

71inline void shm_ring_init(SharedRingHeader *header, uint64_t frame_count,
72 uint64_t element_size) {
73 if (!header || frame_count < 2 || element_size == 0 ||
74 (element_size % 8) != 0)
75 return;
76 uint64_t raw = (frame_count - 1) * arch::PAGE_SIZE;
77 // Round down to a power of two.
78 uint64_t cap = 1;
79 while (cap * 2 <= raw)
80 cap *= 2;
81 __atomic_store_n(&header->magic, SHM_RING_MAGIC, __ATOMIC_RELEASE);
82 __atomic_store_n(&header->version, SHM_RING_VERSION, __ATOMIC_RELEASE);
83 __atomic_store_n(&header->capacity, cap, __ATOMIC_RELEASE);
84 __atomic_store_n(&header->element_size, element_size, __ATOMIC_RELEASE);
85 __atomic_store_n(&header->head, 0, __ATOMIC_RELEASE);
86 __atomic_store_n(&header->tail, 0, __ATOMIC_RELEASE);
87}

90inline bool shm_ring_valid(const SharedRingHeader *header) {
91 return header &&
92 __atomic_load_n(&header->magic, __ATOMIC_ACQUIRE) == SHM_RING_MAGIC &&
93 __atomic_load_n(&header->version, __ATOMIC_ACQUIRE) ==
94 SHM_RING_VERSION;
95}

98inline bool shm_ring_has_space(const SharedRingHeader *header) {
99 if (!header)
100 return false;
101 uint64_t cap = __atomic_load_n(&header->capacity, __ATOMIC_ACQUIRE);
102 uint64_t esz = __atomic_load_n(&header->element_size, __ATOMIC_ACQUIRE);
103 if (cap == 0 || esz == 0)
104 return false;
105 uint64_t h = __atomic_load_n(&header->head, __ATOMIC_RELAXED);
106 uint64_t t = __atomic_load_n(&header->tail, __ATOMIC_ACQUIRE);
107 uint64_t used = (h >= t) ? (h - t) : (h + cap - t);
108 return used + esz <= cap;
109}

112inline bool shm_ring_has_data(const SharedRingHeader *header) {
113 if (!header)
114 return false;
115 uint64_t h = __atomic_load_n(&header->head, __ATOMIC_ACQUIRE);
116 uint64_t t = __atomic_load_n(&header->tail, __ATOMIC_RELAXED);
117 return h != t;
118}

122inline uint8_t *shm_ring_elem(const SharedRingHeader *header,
123 uint8_t *data_base, uint64_t idx) {
124 uint64_t cap = __atomic_load_n(&header->capacity, __ATOMIC_ACQUIRE);
125 uint64_t esz = __atomic_load_n(&header->element_size, __ATOMIC_ACQUIRE);
126 uint64_t off = (idx * esz) & (cap - 1);
127 return data_base + off;
128}

133inline bool shm_ring_push(SharedRingHeader *header, uint8_t *data_base,
134 const uint8_t *src) {
135 if (!header || !data_base || !src)
136 return false;
137 if (!shm_ring_has_space(header))
138 return false;
139 uint64_t esz = __atomic_load_n(&header->element_size, __ATOMIC_ACQUIRE);
140 uint64_t h = __atomic_load_n(&header->head, __ATOMIC_RELAXED);
141 uint8_t *dst = shm_ring_elem(header, data_base, h / esz);
142 for (uint64_t i = 0; i < esz; ++i)
143 dst[i] = src[i];
144 __atomic_store_n(&header->head, h + esz, __ATOMIC_RELEASE);
145 return true;
146}
147
151inline bool shm_ring_pop(SharedRingHeader *header, uint8_t *data_base,
152 uint8_t *dst) {
153 if (!header || !data_base || !dst)
154 return false;
155 if (!shm_ring_has_data(header))
156 return false;
157 uint64_t esz = __atomic_load_n(&header->element_size, __ATOMIC_ACQUIRE);
158 uint64_t t = __atomic_load_n(&header->tail, __ATOMIC_RELAXED);
159 const uint8_t *src = shm_ring_elem(header, data_base, t / esz);
160 for (uint64_t i = 0; i < esz; ++i)
161 dst[i] = src[i];
162 __atomic_store_n(&header->tail, t + esz, __ATOMIC_RELEASE);
163 return true;
164}
