54// Read the current stack pointer (moved byte-identical from scheduler.cpp,
55// issue #323 switch strand: shared by the switch engine's owner-resolution
56// and the remaining scheduler.cpp call sites). Portable across arches.
57inline uint64_t current_sp() noexcept {
58#if defined(CONFIG_ARCH_X86_64)
59 uint64_t sp{};

61 return sp;
62#elif defined(CONFIG_ARCH_AARCH64)
63 uint64_t sp{};

65 return sp;
66#elif defined(CONFIG_ARCH_RISCV64)
67 uint64_t sp{};
68 asm volatile("mv %0, sp" : "=r"(sp));
69 return sp;
70#else
71 return 0; // Unknown arch: no SP readable (counted as null, not an error).
