35// Stored-waiter identity: pointer AND generation conjunction (defeats ABA
36// when a TCB slot is recycled onto a new task). Cheap-pointer-first order.
37// Generations ride as uint64_t (widening-only: Notify stores its gen in a
38// uint64_t field, the arrays hold uint32_t — both directions are exact).
39inline bool same_waiter(const TaskControlBlock *stored, uint64_t stored_gen,
40 const TaskControlBlock &task) noexcept {
41 return stored == &task && stored_gen == task.generation;
42}
43
44// Wakeable waiter: identity holds AND the task is not dead. A cleaned-up
45// task is REAPED, not TERMINATED — both are unwakeable (waking a freed TCB
46// is ready-queue corruption / UAF). The null check is fail-closed: dense
47// waiter arrays never hold null, so it never fires on valid states.
48inline bool waiter_awakeable(const TaskControlBlock *stored,
49 uint64_t stored_gen) noexcept {
50 return stored != nullptr && stored->generation == stored_gen &&
51 stored->state != TaskState::TERMINATED &&
52 stored->state != TaskState::REAPED;
53}
54
55// Stored last-peer staleness for the PIP boost/restore paths: a REAPED or
56// recycled peer must never be touched. Unlike waiter_awakeable, a
57// TERMINATED (not yet reaped) peer is still allocated and safe to boost —
58// hence the narrower check. Pure check only; callers keep their null +
59// holder reset. Callers guarantee non-null (early return above).
60inline bool peer_stale(const TaskControlBlock *stored,
61 uint64_t stored_gen) noexcept {
62 return stored->state == TaskState::REAPED ||
63 stored->generation != stored_gen;
64}
