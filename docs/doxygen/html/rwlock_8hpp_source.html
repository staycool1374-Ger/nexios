53 void read_lock() noexcept {
54 for (;;) {
55 uint32_t seen =
56 atomic_load(&state_, __ATOMIC_ACQUIRE);
57 if ((seen & (k_writer | k_pending)) != 0) {
58 arch::pause();
59 continue;
60 }
61 if (atomic_compare_exchange(&state_, &seen, seen + 1,
62 __ATOMIC_ACQUIRE)) {
63 return;
64 }
65 arch::pause();
66 }
67 }
68
70 void read_unlock() noexcept {
71 atomic_fetch_sub(&state_, 1u, __ATOMIC_RELEASE);
72 }
73
76 void write_lock() noexcept {
77 for (;;) {
78 uint32_t seen =
79 atomic_load(&state_, __ATOMIC_ACQUIRE);
80 if ((seen & k_writer) != 0) {
81 arch::pause();
82 continue;
83 }
84 atomic_fetch_or(&state_, k_pending, __ATOMIC_ACQ_REL);
85 seen = atomic_load(&state_, __ATOMIC_ACQUIRE);
86 if (seen == k_pending &&
87 atomic_compare_exchange(&state_, &seen, k_writer,
88 __ATOMIC_ACQUIRE)) {
89 return;
90 }
91 arch::pause();
92 }
93 }
94
97 void write_unlock() noexcept {
98 atomic_fetch_and(&state_, ~(k_writer | k_pending),
99 __ATOMIC_RELEASE);
100 }
101
104 bool try_read_lock() noexcept {
105 uint32_t seen = atomic_load(&state_, __ATOMIC_ACQUIRE);
106 if ((seen & (k_writer | k_pending)) != 0)
107 return false;
108 return atomic_compare_exchange(&state_, &seen, seen + 1,
109 __ATOMIC_ACQUIRE);
110 }
111
114 bool try_write_lock() noexcept {
115 uint32_t seen = atomic_load(&state_, __ATOMIC_ACQUIRE);
116 if (seen != 0)
117 return false;
118 return atomic_compare_exchange(&state_, &seen, k_writer,
119 __ATOMIC_ACQUIRE);
120 }
121
124 void reset() noexcept {
125 atomic_store(&state_, 0u, __ATOMIC_RELAXED);
126 }
127
128 private:
129 static constexpr uint32_t k_writer = 1u << 31;
130 static constexpr uint32_t k_pending = 1u << 30;
131 uint32_t state_ = 0;
132};
