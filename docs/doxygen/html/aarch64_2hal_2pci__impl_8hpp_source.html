52inline uint64_t pci_ecam_addr(PciBdf bdf, uint8_t reg) {
53 return (arch::HHDM_OFFSET + PCI_ECAM_BASE) |
54 (static_cast<uint64_t>(bdf.bus) << 20) |
55 (static_cast<uint64_t>(bdf.device) << 15) |
56 (static_cast<uint64_t>(bdf.function) << 12) | reg;
57}
