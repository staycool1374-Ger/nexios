36// QEMU virt fw_cfg MMIO base: per-arch (DTB-proven — the aarch64 and
37// riscv64 virt machines place it differently).
38// aarch64 `virt`: fw-cfg@9020000. riscv64 `virt`: fw-cfg@10100000,
39// inside the UART's 2MB Device leaf (no extra boot leaf needed).
40#if defined(CONFIG_ARCH_AARCH64)
41inline constexpr uint64_t FW_CFG_MMIO_BASE = 0x09020000ULL;
42#else
43inline constexpr uint64_t FW_CFG_MMIO_BASE = 0x10100000ULL;
44#endif
45// Register offsets (MMIO layout).
46inline constexpr uint64_t FW_CFG_DATA = 0x00; // u8 PIO data (advances)
47inline constexpr uint64_t FW_CFG_SELECTOR = 0x08; // u16 BE file selector
48inline constexpr uint64_t FW_CFG_DMA = 0x10; // u64 BE DMA address
49// Well-known selectors.
50inline constexpr uint16_t FW_CFG_SIG = 0x0000; // "QEMU" (4 bytes)
51inline constexpr uint16_t FW_CFG_FILE_DIR = 0x0019; // file directory
52// DMA control bits.
53inline constexpr uint32_t FW_CFG_DMA_ERROR = 1U << 0;
54inline constexpr uint32_t FW_CFG_DMA_SELECT = 1U << 3;
55inline constexpr uint32_t FW_CFG_DMA_WRITE = 1U << 4;
56// Bounds.
57inline constexpr uint32_t FW_CFG_MAX_DIR_FILES = 128;
58inline constexpr int FW_CFG_DMA_TIMEOUT_ITERS = 1000000;
59
62inline uint64_t fw_cfg_virt() {
63#if defined(CONFIG_ARCH_AARCH64)
64 return kernel::HHDM_OFFSET + FW_CFG_MMIO_BASE;
65#elif defined(CONFIG_ARCH_RISCV64)
66 return FW_CFG_MMIO_BASE; // identity L0[0] window covers MMIO
67#else
68 return 0;
69#endif
70}

73inline constexpr uint16_t fw_cfg_be16(uint16_t v) {
74 return static_cast<uint16_t>(((v & 0xFFU) << 8) | ((v >> 8) & 0xFFU));
75}
76inline constexpr uint32_t fw_cfg_be32(uint32_t v) {
77 return __builtin_bswap32(v);
78}
79inline constexpr uint64_t fw_cfg_be64(uint64_t v) {
80 return __builtin_bswap64(v);
81}
