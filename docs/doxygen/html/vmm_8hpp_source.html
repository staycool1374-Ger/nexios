599 static constexpr uint64_t PAGE_TABLE_ENTRIES = 512;
604 static constexpr uint64_t PAGE_TABLE_ADDR_MASK = ~0xFFFULL;
605 static_assert((PAGE_TABLE_ADDR_MASK | 0xFFFULL) == ~0ULL,
606 "addr mask must preserve all high bits");
607 static_assert((0xFFFULL & PAGE_TABLE_ADDR_MASK) == 0,
608 "addr mask must clear the low 12 bits");
609
610#if defined(CONFIG_ARCH_AARCH64)
611 // aarch64 stage-1 page-table descriptor bit layout.
612 // Table descriptors (L0-L2): bits[1:0]=11 (V=1, bit1=1 → next-level
613 // table). Block descriptors (L1-L2): bits[1:0]=01 (bit1=0).
614 // Page descriptors (L3): bits[1:0]=11 — bit1=0 at L3 is a
615 // block-at-invalid-level encoding and faults as a translation fault.
616 // AP[1:0] in bits[7:6]; 00=EL1-RW/EL0-*, 01=EL1-RW/EL0-RW,
617 // 10=EL1-RO/EL0-*, 11=EL1-RO/EL0-RO.
618 // AF (Access Flag) = bit 10 (stage-1); an AF=0 leaf faults on first
619 // access. Bits 9:8 = SH (shareability).
620 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
621 static constexpr uint64_t PAGE_TABLE = 1ULL << 1;
622 static constexpr uint64_t PAGE_AF = 1ULL << 10; // Access Flag (stage-1)
623 static constexpr uint64_t PAGE_AP_USER = 1ULL
624 << 6; // AP[0] - EL0 accessible
625 static constexpr uint64_t PAGE_AP_RO = 1ULL << 7; // AP[1] - read-only (EL1)
626 // MP-4.4: PXN (bit 53) forbids EL1 execution; UXN (bit 54) forbids EL0
627 // execution. Kernel data pages carry both; user pages carry PXN
628 // (SMEP parity) unless NX adds UXN.
629 static constexpr uint64_t PAGE_PXN = 1ULL << 53;
630 static constexpr uint64_t PAGE_UXN = 1ULL << 54;
634 static constexpr uint64_t PAGE_ATTR_NORMAL = 1ULL << 2;
635
636 // Compatibility aliases (used in flag-building expressions)
637 static constexpr uint64_t PAGE_WRITE = 0; // no-op - writable by default
638 static constexpr uint64_t PAGE_USER = PAGE_AP_USER;
639 static constexpr uint64_t PAGE_HUGE = 0; // not a flag; detected via level
640#elif defined(CONFIG_ARCH_RISCV64)
641 // RISC-V Sv39 PTE format (from privilege spec)
642 // bit 0: V (Valid)
643 // bit 1: R (Readable)
644 // bit 2: W (Writable)
645 // bit 3: X (Executable)
646 // bit 4: U (User)
647 // bit 5: G (Global)
648 // bit 6: A (Accessed)
649 // bit 7: D (Dirty)
650 // bit 8: RSW (reserved for software)
651 // bits 9-53: PPN[0..44] (physical page number)
652 // bits 54-63: reserved for future use
653 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0; // V
654 static constexpr uint64_t PAGE_READ = 1ULL << 1; // R
655 static constexpr uint64_t PAGE_WRITE = 1ULL << 2; // W
656 static constexpr uint64_t PAGE_EXEC = 1ULL << 3; // X
657 static constexpr uint64_t PAGE_USER = 1ULL << 4; // U
658 static constexpr uint64_t PAGE_GLOBAL = 1ULL << 5; // G
659 static constexpr uint64_t PAGE_ACCESSED = 1ULL << 6; // A
660 static constexpr uint64_t PAGE_DIRTY = 1ULL << 7; // D
661 // Table entry: V=1, R=W=X=0 (points to next level)
662 static constexpr uint64_t PAGE_TABLE =
663 PAGE_PRESENT; // V=1, no RWX = table pointer
664 // Block/leaf detection: if V=1 and (R|W|X)=1 then leaf page
665 static constexpr uint64_t PAGE_HUGE = 0; // detected via R|W|X bits
666#else
667 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
668 static constexpr uint64_t PAGE_WRITE = 1ULL << 1;
669 static constexpr uint64_t PAGE_USER = 1ULL << 2;
670 static constexpr uint64_t PAGE_HUGE = 1ULL << 7;
671 // NX lives in the x86_64 HAL (arch::ArchPageTable::NX, issue #363 P3):
672 // it is an x86-PTE encoding bit, not VMM vocabulary.
673#endif
674
675#if defined(CONFIG_ARCH_AARCH64)
676 // aarch64: physical frame bits [47:12] (48-bit PA). Masks off flags,
677 // UXN (bit 54) and PXN (bit 53).
678 static constexpr uint64_t PAGE_FRAME_MASK = 0x0000FFFFFFFFF000ULL;
679 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x0000FFFFFFE00000ULL;
680#elif defined(CONFIG_ARCH_RISCV64)
681 // Sv39: PPN bits [43:12] (34-bit PA). Masks off flags + reserved bits.
682 static constexpr uint64_t PAGE_FRAME_MASK = 0x00000003FFFFFFF000ULL;
683 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x00000003FFFFE00000ULL;
684 // Issue #206: the masks above are x86-domain (phys & mask) and WRONG for
685 // Sv39 PTEs, which carry (PPN << 10). Use the shift-based helpers in
686 // the public section below for every Sv39 PTE address codec; the masks
687 // stay for x86/aarch64 and for documentation of the PA width (1TB cap).
688#else
689 // x86_64: physical frame bits [51:12] (52-bit PA). Masks off all flags
690 // INCLUDING NX (bit 63) — without this, a leaf PTE with NX set yields a
691 // "physical address" whose bit 63 wraps HHDM_OFFSET + phys to a
692 // non-canonical address (VULN-H1 GPF).
693 static constexpr uint64_t PAGE_FRAME_MASK = 0x000FFFFFFFFFF000ULL;
694 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x000FFFFFFFE00000ULL;
695#endif
