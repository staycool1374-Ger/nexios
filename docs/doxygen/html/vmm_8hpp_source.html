260#if defined(CONFIG_ARCH_AARCH64)
261 // aarch64 stage-1 page-table descriptor bit layout.
262 // Table descriptors (L0-L2): bits[1:0]=11 (V=1, bit1=1 → next-level
263 // table). Block descriptors (L1-L2): bits[1:0]=01 (bit1=0).
264 // Page descriptors (L3): bits[1:0]=11 — bit1=0 at L3 is a
265 // block-at-invalid-level encoding and faults as a translation fault.
266 // AP[1:0] in bits[7:6]; 00=EL1-RW/EL0-*, 01=EL1-RW/EL0-RW,
267 // 10=EL1-RO/EL0-*, 11=EL1-RO/EL0-RO.
268 // AF (Access Flag) = bit 10 (stage-1); an AF=0 leaf faults on first
269 // access. Bits 9:8 = SH (shareability).
270 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
271 static constexpr uint64_t PAGE_TABLE = 1ULL << 1;
272 static constexpr uint64_t PAGE_AF = 1ULL << 10; // Access Flag (stage-1)
273 static constexpr uint64_t PAGE_AP_USER = 1ULL
274 << 6; // AP[0] - EL0 accessible
275 static constexpr uint64_t PAGE_AP_RO = 1ULL << 7; // AP[1] - read-only (EL1)
276 // MP-4.4: PXN (bit 53) forbids EL1 execution; UXN (bit 54) forbids EL0
277 // execution. Kernel data pages carry both; user pages carry PXN
278 // (SMEP parity) unless NX adds UXN.
279 static constexpr uint64_t PAGE_PXN = 1ULL << 53;
280 static constexpr uint64_t PAGE_UXN = 1ULL << 54;
284 static constexpr uint64_t PAGE_ATTR_NORMAL = 1ULL << 2;
285
286 // Compatibility aliases (used in flag-building expressions)
287 static constexpr uint64_t PAGE_WRITE = 0; // no-op - writable by default
288 static constexpr uint64_t PAGE_USER = PAGE_AP_USER;
289 static constexpr uint64_t PAGE_HUGE = 0; // not a flag; detected via level
290#elif defined(CONFIG_ARCH_RISCV64)
291 // RISC-V Sv39 PTE format (from privilege spec)
292 // bit 0: V (Valid)
293 // bit 1: R (Readable)
294 // bit 2: W (Writable)
295 // bit 3: X (Executable)
296 // bit 4: U (User)
297 // bit 5: G (Global)
298 // bit 6: A (Accessed)
299 // bit 7: D (Dirty)
300 // bit 8: RSW (reserved for software)
301 // bits 9-53: PPN[0..44] (physical page number)
302 // bits 54-63: reserved for future use
303 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0; // V
304 static constexpr uint64_t PAGE_READ = 1ULL << 1; // R
305 static constexpr uint64_t PAGE_WRITE = 1ULL << 2; // W
306 static constexpr uint64_t PAGE_EXEC = 1ULL << 3; // X
307 static constexpr uint64_t PAGE_USER = 1ULL << 4; // U
308 static constexpr uint64_t PAGE_GLOBAL = 1ULL << 5; // G
309 static constexpr uint64_t PAGE_ACCESSED = 1ULL << 6; // A
310 static constexpr uint64_t PAGE_DIRTY = 1ULL << 7; // D
311 // Table entry: V=1, R=W=X=0 (points to next level)
312 static constexpr uint64_t PAGE_TABLE =
313 PAGE_PRESENT; // V=1, no RWX = table pointer
314 // Block/leaf detection: if V=1 and (R|W|X)=1 then leaf page
315 static constexpr uint64_t PAGE_HUGE = 0; // detected via R|W|X bits
316#else
317 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
318 static constexpr uint64_t PAGE_WRITE = 1ULL << 1;
319 static constexpr uint64_t PAGE_USER = 1ULL << 2;
320 static constexpr uint64_t PAGE_HUGE = 1ULL << 7;
321#endif
322
323#if defined(CONFIG_ARCH_AARCH64)
324 // aarch64: physical frame bits [47:12] (48-bit PA). Masks off flags,
325 // UXN (bit 54) and PXN (bit 53).
326 static constexpr uint64_t PAGE_FRAME_MASK = 0x0000FFFFFFFFF000ULL;
327 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x0000FFFFFFE00000ULL;
328#elif defined(CONFIG_ARCH_RISCV64)
329 // Sv39: PPN bits [43:12] (34-bit PA). Masks off flags + reserved bits.
330 static constexpr uint64_t PAGE_FRAME_MASK = 0x00000003FFFFFFF000ULL;
331 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x00000003FFFFE00000ULL;
332 // Issue #206: the masks above are x86-domain (phys & mask) and WRONG for
333 // Sv39 PTEs, which carry (PPN << 10). Use the shift-based helpers below
334 // for every Sv39 PTE address codec; the masks stay for x86/aarch64 and
335 // for documentation of the PA width (1TB cap).
336 static inline uint64_t sv39_pte_phys(uint64_t pte) {
337 return ((pte >> 10) << 12) & 0xFFFFFFFFFFF000ULL;
338 }
339 static inline uint64_t sv39_phys_pte(uint64_t phys) {
340 return (phys >> 12) << 10;
341 }
342 static inline bool sv39_is_leaf(uint64_t pte) {
343 return (pte & 0xEU) != 0; // R|W|X set = leaf (table: V only)
344 }
345#else
346 // x86_64: physical frame bits [51:12] (52-bit PA). Masks off all flags
347 // INCLUDING NX (bit 63) — without this, a leaf PTE with NX set yields a
348 // "physical address" whose bit 63 wraps HHDM_OFFSET + phys to a
349 // non-canonical address (VULN-H1 GPF).
350 static constexpr uint64_t PAGE_FRAME_MASK = 0x000FFFFFFFFFF000ULL;
351 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x000FFFFFFFE00000ULL;
352#endif
353
354#if defined(CONFIG_ARCH_RISCV64)
355 // Sv39: 3-level page table
356 // L0 (1GB): bits 38:30
357 // L1 (2MB): bits 29:21
358 // L2 (4KB): bits 20:12
359 static constexpr uint64_t L0_SHIFT = 30;
360 static constexpr uint64_t L1_SHIFT = 21;
361 static constexpr uint64_t L2_SHIFT = 12;
362 static constexpr uint64_t TABLE_MASK = 0x1FFULL;
363 static constexpr uint64_t L0_MASK = TABLE_MASK << L0_SHIFT;
364 static constexpr uint64_t L1_MASK = TABLE_MASK << L1_SHIFT;
365 static constexpr uint64_t L2_MASK = TABLE_MASK << L2_SHIFT;
366#endif
