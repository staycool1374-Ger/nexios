583#if defined(CONFIG_ARCH_AARCH64)
584 // aarch64 stage-1 page-table descriptor bit layout.
585 // Table descriptors (L0-L2): bits[1:0]=11 (V=1, bit1=1 → next-level
586 // table). Block descriptors (L1-L2): bits[1:0]=01 (bit1=0).
587 // Page descriptors (L3): bits[1:0]=11 — bit1=0 at L3 is a
588 // block-at-invalid-level encoding and faults as a translation fault.
589 // AP[1:0] in bits[7:6]; 00=EL1-RW/EL0-*, 01=EL1-RW/EL0-RW,
590 // 10=EL1-RO/EL0-*, 11=EL1-RO/EL0-RO.
591 // AF (Access Flag) = bit 10 (stage-1); an AF=0 leaf faults on first
592 // access. Bits 9:8 = SH (shareability).
593 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
594 static constexpr uint64_t PAGE_TABLE = 1ULL << 1;
595 static constexpr uint64_t PAGE_AF = 1ULL << 10; // Access Flag (stage-1)
596 static constexpr uint64_t PAGE_AP_USER = 1ULL
597 << 6; // AP[0] - EL0 accessible
598 static constexpr uint64_t PAGE_AP_RO = 1ULL << 7; // AP[1] - read-only (EL1)
599 // MP-4.4: PXN (bit 53) forbids EL1 execution; UXN (bit 54) forbids EL0
600 // execution. Kernel data pages carry both; user pages carry PXN
601 // (SMEP parity) unless NX adds UXN.
602 static constexpr uint64_t PAGE_PXN = 1ULL << 53;
603 static constexpr uint64_t PAGE_UXN = 1ULL << 54;
607 static constexpr uint64_t PAGE_ATTR_NORMAL = 1ULL << 2;
608
609 // Compatibility aliases (used in flag-building expressions)
610 static constexpr uint64_t PAGE_WRITE = 0; // no-op - writable by default
611 static constexpr uint64_t PAGE_USER = PAGE_AP_USER;
612 static constexpr uint64_t PAGE_HUGE = 0; // not a flag; detected via level
613#elif defined(CONFIG_ARCH_RISCV64)
614 // RISC-V Sv39 PTE format (from privilege spec)
615 // bit 0: V (Valid)
616 // bit 1: R (Readable)
617 // bit 2: W (Writable)
618 // bit 3: X (Executable)
619 // bit 4: U (User)
620 // bit 5: G (Global)
621 // bit 6: A (Accessed)
622 // bit 7: D (Dirty)
623 // bit 8: RSW (reserved for software)
624 // bits 9-53: PPN[0..44] (physical page number)
625 // bits 54-63: reserved for future use
626 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0; // V
627 static constexpr uint64_t PAGE_READ = 1ULL << 1; // R
628 static constexpr uint64_t PAGE_WRITE = 1ULL << 2; // W
629 static constexpr uint64_t PAGE_EXEC = 1ULL << 3; // X
630 static constexpr uint64_t PAGE_USER = 1ULL << 4; // U
631 static constexpr uint64_t PAGE_GLOBAL = 1ULL << 5; // G
632 static constexpr uint64_t PAGE_ACCESSED = 1ULL << 6; // A
633 static constexpr uint64_t PAGE_DIRTY = 1ULL << 7; // D
634 // Table entry: V=1, R=W=X=0 (points to next level)
635 static constexpr uint64_t PAGE_TABLE =
636 PAGE_PRESENT; // V=1, no RWX = table pointer
637 // Block/leaf detection: if V=1 and (R|W|X)=1 then leaf page
638 static constexpr uint64_t PAGE_HUGE = 0; // detected via R|W|X bits
639#else
640 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
641 static constexpr uint64_t PAGE_WRITE = 1ULL << 1;
642 static constexpr uint64_t PAGE_USER = 1ULL << 2;
643 static constexpr uint64_t PAGE_HUGE = 1ULL << 7;
644 // NX lives in the x86_64 HAL (arch::ArchPageTable::NX, issue #363 P3):
645 // it is an x86-PTE encoding bit, not VMM vocabulary.
646#endif
647
648#if defined(CONFIG_ARCH_AARCH64)
649 // aarch64: physical frame bits [47:12] (48-bit PA). Masks off flags,
650 // UXN (bit 54) and PXN (bit 53).
651 static constexpr uint64_t PAGE_FRAME_MASK = 0x0000FFFFFFFFF000ULL;
652 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x0000FFFFFFE00000ULL;
653#elif defined(CONFIG_ARCH_RISCV64)
654 // Sv39: PPN bits [43:12] (34-bit PA). Masks off flags + reserved bits.
655 static constexpr uint64_t PAGE_FRAME_MASK = 0x00000003FFFFFFF000ULL;
656 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x00000003FFFFE00000ULL;
657 // Issue #206: the masks above are x86-domain (phys & mask) and WRONG for
658 // Sv39 PTEs, which carry (PPN << 10). Use the shift-based helpers in
659 // the public section below for every Sv39 PTE address codec; the masks
660 // stay for x86/aarch64 and for documentation of the PA width (1TB cap).
661#else
662 // x86_64: physical frame bits [51:12] (52-bit PA). Masks off all flags
663 // INCLUDING NX (bit 63) — without this, a leaf PTE with NX set yields a
664 // "physical address" whose bit 63 wraps HHDM_OFFSET + phys to a
665 // non-canonical address (VULN-H1 GPF).
666 static constexpr uint64_t PAGE_FRAME_MASK = 0x000FFFFFFFFFF000ULL;
667 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x000FFFFFFFE00000ULL;
668#endif
