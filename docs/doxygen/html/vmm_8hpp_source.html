196#if defined(CONFIG_ARCH_AARCH64)
197 // aarch64 stage-1 page-table descriptor bit layout.
198 // Table descriptors (L0-L2) and page descriptors (L3): bits[1:0]=11.
199 // Block descriptors (L1-L2): bits[1:0]=01.
200 // AP[1:0] in bits[7:6]; 00=EL1-RW/EL0-*, 01=EL1-RW/EL0-RW,
201 // 10=EL1-RO/EL0-*, 11=EL1-RO/EL0-RO.
202 // AF (Access Flag) must be 1 for stage-1 - bit 8 (NOT bit 10).
203 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
204 static constexpr uint64_t PAGE_TABLE = 1ULL << 1;
205 static constexpr uint64_t PAGE_AF = 1ULL << 8; // Access Flag (stage-1)
206 static constexpr uint64_t PAGE_AP_USER = 1ULL
207 << 6; // AP[0] - EL0 accessible
208 static constexpr uint64_t PAGE_AP_RO = 1ULL << 7; // AP[1] - read-only (EL1)
209
210 // Compatibility aliases (used in flag-building expressions)
211 static constexpr uint64_t PAGE_WRITE = 0; // no-op - writable by default
212 static constexpr uint64_t PAGE_USER = PAGE_AP_USER;
213 static constexpr uint64_t PAGE_HUGE = 0; // not a flag; detected via level
214#elif defined(CONFIG_ARCH_RISCV64)
215 // RISC-V Sv39 PTE format (from privilege spec)
216 // bit 0: V (Valid)
217 // bit 1: R (Readable)
218 // bit 2: W (Writable)
219 // bit 3: X (Executable)
220 // bit 4: U (User)
221 // bit 5: G (Global)
222 // bit 6: A (Accessed)
223 // bit 7: D (Dirty)
224 // bit 8: RSW (reserved for software)
225 // bits 9-53: PPN[0..44] (physical page number)
226 // bits 54-63: reserved for future use
227 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0; // V
228 static constexpr uint64_t PAGE_READ = 1ULL << 1; // R
229 static constexpr uint64_t PAGE_WRITE = 1ULL << 2; // W
230 static constexpr uint64_t PAGE_EXEC = 1ULL << 3; // X
231 static constexpr uint64_t PAGE_USER = 1ULL << 4; // U
232 static constexpr uint64_t PAGE_GLOBAL = 1ULL << 5; // G
233 static constexpr uint64_t PAGE_ACCESSED = 1ULL << 6; // A
234 static constexpr uint64_t PAGE_DIRTY = 1ULL << 7; // D
235 // Table entry: V=1, R=W=X=0 (points to next level)
236 static constexpr uint64_t PAGE_TABLE =
237 PAGE_PRESENT; // V=1, no RWX = table pointer
238 // Block/leaf detection: if V=1 and (R|W|X)=1 then leaf page
239 static constexpr uint64_t PAGE_HUGE = 0; // detected via R|W|X bits
240#else
241 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
242 static constexpr uint64_t PAGE_WRITE = 1ULL << 1;
243 static constexpr uint64_t PAGE_USER = 1ULL << 2;
244 static constexpr uint64_t PAGE_HUGE = 1ULL << 7;
245#endif
246
247#if defined(CONFIG_ARCH_AARCH64)
248 // aarch64: physical frame bits [47:12] (48-bit PA). Masks off flags,
249 // UXN (bit 54) and PXN (bit 55).
250 static constexpr uint64_t PAGE_FRAME_MASK = 0x0000FFFFFFFFF000ULL;
251 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x0000FFFFFFE00000ULL;
252#elif defined(CONFIG_ARCH_RISCV64)
253 // Sv39: PPN bits [43:12] (34-bit PA). Masks off flags + reserved bits.
254 static constexpr uint64_t PAGE_FRAME_MASK = 0x00000003FFFFFFF000ULL;
255 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x00000003FFFFE00000ULL;
256#else
257 // x86_64: physical frame bits [51:12] (52-bit PA). Masks off all flags
258 // INCLUDING NX (bit 63) — without this, a leaf PTE with NX set yields a
259 // "physical address" whose bit 63 wraps HHDM_OFFSET + phys to a
260 // non-canonical address (VULN-H1 GPF).
261 static constexpr uint64_t PAGE_FRAME_MASK = 0x000FFFFFFFFFF000ULL;
262 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x000FFFFFFFE00000ULL;
263#endif
264
265#if defined(CONFIG_ARCH_RISCV64)
266 // Sv39: 3-level page table
267 // L0 (1GB): bits 38:30
268 // L1 (2MB): bits 29:21

275 static constexpr uint64_t L1_MASK = TABLE_MASK << L1_SHIFT;

277#endif
