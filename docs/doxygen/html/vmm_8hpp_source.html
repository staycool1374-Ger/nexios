262#if defined(CONFIG_ARCH_AARCH64)
263 // aarch64 stage-1 page-table descriptor bit layout.
264 // Table descriptors (L0-L2): bits[1:0]=11 (V=1, bit1=1 → next-level
265 // table). Block descriptors (L1-L2): bits[1:0]=01 (bit1=0).
266 // Page descriptors (L3): bits[1:0]=11 — bit1=0 at L3 is a
267 // block-at-invalid-level encoding and faults as a translation fault.
268 // AP[1:0] in bits[7:6]; 00=EL1-RW/EL0-*, 01=EL1-RW/EL0-RW,
269 // 10=EL1-RO/EL0-*, 11=EL1-RO/EL0-RO.
270 // AF (Access Flag) = bit 10 (stage-1); an AF=0 leaf faults on first
271 // access. Bits 9:8 = SH (shareability).
272 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
273 static constexpr uint64_t PAGE_TABLE = 1ULL << 1;
274 static constexpr uint64_t PAGE_AF = 1ULL << 10; // Access Flag (stage-1)
275 static constexpr uint64_t PAGE_AP_USER = 1ULL
276 << 6; // AP[0] - EL0 accessible
277 static constexpr uint64_t PAGE_AP_RO = 1ULL << 7; // AP[1] - read-only (EL1)
278 // MP-4.4: PXN (bit 53) forbids EL1 execution; UXN (bit 54) forbids EL0
279 // execution. Kernel data pages carry both; user pages carry PXN
280 // (SMEP parity) unless NX adds UXN.
281 static constexpr uint64_t PAGE_PXN = 1ULL << 53;
282 static constexpr uint64_t PAGE_UXN = 1ULL << 54;
286 static constexpr uint64_t PAGE_ATTR_NORMAL = 1ULL << 2;
287
288 // Compatibility aliases (used in flag-building expressions)
289 static constexpr uint64_t PAGE_WRITE = 0; // no-op - writable by default
290 static constexpr uint64_t PAGE_USER = PAGE_AP_USER;
291 static constexpr uint64_t PAGE_HUGE = 0; // not a flag; detected via level
292#elif defined(CONFIG_ARCH_RISCV64)
293 // RISC-V Sv39 PTE format (from privilege spec)
294 // bit 0: V (Valid)
295 // bit 1: R (Readable)
296 // bit 2: W (Writable)
297 // bit 3: X (Executable)
298 // bit 4: U (User)
299 // bit 5: G (Global)
300 // bit 6: A (Accessed)
301 // bit 7: D (Dirty)
302 // bit 8: RSW (reserved for software)
303 // bits 9-53: PPN[0..44] (physical page number)
304 // bits 54-63: reserved for future use
305 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0; // V
306 static constexpr uint64_t PAGE_READ = 1ULL << 1; // R
307 static constexpr uint64_t PAGE_WRITE = 1ULL << 2; // W
308 static constexpr uint64_t PAGE_EXEC = 1ULL << 3; // X
309 static constexpr uint64_t PAGE_USER = 1ULL << 4; // U
310 static constexpr uint64_t PAGE_GLOBAL = 1ULL << 5; // G
311 static constexpr uint64_t PAGE_ACCESSED = 1ULL << 6; // A
312 static constexpr uint64_t PAGE_DIRTY = 1ULL << 7; // D
313 // Table entry: V=1, R=W=X=0 (points to next level)
314 static constexpr uint64_t PAGE_TABLE =
315 PAGE_PRESENT; // V=1, no RWX = table pointer
316 // Block/leaf detection: if V=1 and (R|W|X)=1 then leaf page
317 static constexpr uint64_t PAGE_HUGE = 0; // detected via R|W|X bits
318#else
319 static constexpr uint64_t PAGE_PRESENT = 1ULL << 0;
320 static constexpr uint64_t PAGE_WRITE = 1ULL << 1;
321 static constexpr uint64_t PAGE_USER = 1ULL << 2;
322 static constexpr uint64_t PAGE_HUGE = 1ULL << 7;
323#endif
324
325#if defined(CONFIG_ARCH_AARCH64)
326 // aarch64: physical frame bits [47:12] (48-bit PA). Masks off flags,
327 // UXN (bit 54) and PXN (bit 53).
328 static constexpr uint64_t PAGE_FRAME_MASK = 0x0000FFFFFFFFF000ULL;
329 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x0000FFFFFFE00000ULL;
330#elif defined(CONFIG_ARCH_RISCV64)
331 // Sv39: PPN bits [43:12] (34-bit PA). Masks off flags + reserved bits.
332 static constexpr uint64_t PAGE_FRAME_MASK = 0x00000003FFFFFFF000ULL;
333 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x00000003FFFFE00000ULL;
334#else
335 // x86_64: physical frame bits [51:12] (52-bit PA). Masks off all flags
336 // INCLUDING NX (bit 63) — without this, a leaf PTE with NX set yields a
337 // "physical address" whose bit 63 wraps HHDM_OFFSET + phys to a
338 // non-canonical address (VULN-H1 GPF).
339 static constexpr uint64_t PAGE_FRAME_MASK = 0x000FFFFFFFFFF000ULL;
340 static constexpr uint64_t PAGE_HUGE_FRAME_MASK = 0x000FFFFFFFE00000ULL;
341#endif
342
343#if defined(CONFIG_ARCH_RISCV64)
344 // Sv39: 3-level page table
345 // L0 (1GB): bits 38:30
346 // L1 (2MB): bits 29:21
347 // L2 (4KB): bits 20:12
348 static constexpr uint64_t L0_SHIFT = 30;
349 static constexpr uint64_t L1_SHIFT = 21;
350 static constexpr uint64_t L2_SHIFT = 12;
351 static constexpr uint64_t TABLE_MASK = 0x1FFULL;
352 static constexpr uint64_t L0_MASK = TABLE_MASK << L0_SHIFT;
353 static constexpr uint64_t L1_MASK = TABLE_MASK << L1_SHIFT;
354 static constexpr uint64_t L2_MASK = TABLE_MASK << L2_SHIFT;
355#endif
