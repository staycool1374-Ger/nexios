134constexpr __uint128_t syscall_fast_mask_from(const SyscallNumber *list,
135 size_t count) {
136 __uint128_t mask = 0;
137 for (size_t i = 0; i < count; ++i) {
138 mask |= ((__uint128_t)1 << static_cast<uint64_t>(list[i]));
139 }
140 return mask;
141}
