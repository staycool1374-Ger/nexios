60class KernelObject {

74 virtual void revoke() noexcept {

84 bool acquire() noexcept {
85 if (__atomic_load_n(&revoked_, __ATOMIC_ACQUIRE))
86 return false;
87 __atomic_fetch_add(&ref_count_, 1U, __ATOMIC_RELAXED);
88 return true;
89 }
90
96 void release() noexcept {
97 if (__atomic_fetch_sub(&ref_count_, 1U, __ATOMIC_ACQ_REL) == 1U) {
98 dispose();
99 }
100 }
