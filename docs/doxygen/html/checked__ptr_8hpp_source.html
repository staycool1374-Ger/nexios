41static inline bool is_user_range(const void *user_ptr, uint64_t size) {
42 if (!user_ptr)
43 return false;
44 uint64_t addr = reinterpret_cast<uint64_t>(user_ptr);
45 if (addr >= USER_SPACE_LIMIT)
46 return false;
47 if (size == 0)
48 return true;
49 uint64_t end = addr + size - 1;
50 if (end < addr)
51 return false;
52 if (end >= USER_SPACE_LIMIT)
53 return false;
54 return true;
55}

179static inline bool is_user_string(const void *user_ptr,
180 uint64_t max_len = 4096) {
181 uint64_t addr = reinterpret_cast<uint64_t>(user_ptr);
182 if (addr >= USER_SPACE_LIMIT || addr == 0)
183 return false;
184 uint64_t end = addr + max_len - 1;
185 if (end < addr || end >= USER_SPACE_LIMIT)
186 return false;
187 // Check if the page is actually mapped before dereferencing.
188 // After snapshot_restore clears kernel PML4 user entries, unmapped
189 // user addresses would cause a Page Fault here. Probe the CURRENTLY
190 // ACTIVE page table (the current task's private PML4 under MP-1 — the
191 // boot kernel_pml4_ has its user entries zeroed, so VMM::virt_to_phys
192 // would wrongly report every user string unmapped). virt_to_phys_in_pml4
193 // returns 0 for unmapped pages without faulting.
194 uint64_t probe_cr3 = arch::read_cr3();
195 if (VMM::virt_to_phys_in_pml4(addr, probe_cr3) == 0)
196 return false;
197 g_user_access_recover_ip = reinterpret_cast<uint64_t>(&&recover_str_chk);
198 arch::stac();
199 auto *p = static_cast<const volatile char *>(user_ptr);
200 for (uint64_t i = 0; i < max_len; ++i) {
201 if (p[i] == '\0') {
202 arch::clac();
203 g_user_access_recover_ip = 0;
204 return true;
205 }
206 }
207 arch::clac();
208 g_user_access_recover_ip = 0;
209 return false;
210
211recover_str_chk:
212 arch::clac();
213 g_user_access_recover_ip = 0;
214 return false;
215}

222static inline bool strncpy_from_user(char *dst, const char *src,
223 uint64_t max_len) {
224 if (!is_user_string(src, max_len))
225 return false;
226 g_user_access_recover_ip = reinterpret_cast<uint64_t>(&&recover_str);
227 arch::stac();
228 for (uint64_t i = 0; i < max_len; ++i) {
229 dst[i] = src[i];
230 if (src[i] == '\0') {
231 arch::clac();
232 g_user_access_recover_ip = 0;
233 return true;
234 }
235 }
236 dst[max_len - 1] = '\0';
237 arch::clac();
238 g_user_access_recover_ip = 0;
239 return true;
240
241recover_str:
242 arch::clac();
243 g_user_access_recover_ip = 0;
244 if (max_len > 0)
245 dst[0] = '\0';
246 return false;
247}
