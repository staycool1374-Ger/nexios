71static inline bool is_user_range(const void *user_ptr, uint64_t size) {
72 if (!user_ptr)
73 return false;
74 uint64_t addr = reinterpret_cast<uint64_t>(user_ptr);
75 if (addr >= USER_SPACE_LIMIT)
76 return false;
77 if (size == 0)
78 return true;
79 uint64_t end = addr + size - 1;
80 if (end < addr)
81 return false;
82 if (end >= USER_SPACE_LIMIT)
83 return false;
84 return true;
85}

228static inline bool is_user_string(const void *user_ptr,
229 uint64_t max_len = 4096) {
230 uint64_t addr = reinterpret_cast<uint64_t>(user_ptr);
231 if (addr >= USER_SPACE_LIMIT || addr == 0)
232 return false;
233 uint64_t end = addr + max_len - 1;
234 if (end < addr || end >= USER_SPACE_LIMIT)
235 return false;
236 // Check if the page is actually mapped before dereferencing.
237 // After snapshot_restore clears kernel PML4 user entries, unmapped
238 // user addresses would cause a Page Fault here. Probe the CURRENTLY
239 // ACTIVE page table (the current task's private PML4 under MP-1 — the
240 // boot kernel_pml4_ has its user entries zeroed, so VMM::virt_to_phys
241 // would wrongly report every user string unmapped). virt_to_phys_in_pml4
242 // returns 0 for unmapped pages without faulting.
243 // Issue #171 P6: read_cr3() is a LIVE root — aa64 TTBR0 carries the
244 // ASID in bits [63:48] and the walk mask keeps it, wrapping the L0
245 // base non-canonical (EL1 abort). Untag to a table phys first
246 // (x86/riscv pass through unchanged).
247 uint64_t probe_cr3 = VMM::vmm_arch_root_phys(arch::read_cr3());
248 if (VMM::virt_to_phys_in_pml4(addr, probe_cr3) == 0)
249 return false;
250 // Hoisted above the keep-alive edge: the goto below must not cross a
251 // non-vacuous initialization (issue #149). The cast itself cannot fault.
252 auto *p = static_cast<const volatile char *>(user_ptr);
253 g_user_access_recover_ip = reinterpret_cast<uint64_t>(&&recover_str_chk);
254 NEXIOS_FAULT_RECOVERY_KEEP(recover_str_chk);
255 arch::stac();
256 for (uint64_t i = 0; i < max_len; ++i) {
257 if (p[i] == '\0') {
258 arch::clac();
259 g_user_access_recover_ip = 0;
260 return true;
261 }
262 }
263 arch::clac();
264 g_user_access_recover_ip = 0;
265 return false;
266
267recover_str_chk:
268 arch::clac();
269 g_user_access_recover_ip = 0;
270 return false;
271}

278static inline bool strncpy_from_user(char *dst, const char *src,
279 uint64_t max_len) {
280 if (!is_user_string(src, max_len))
281 return false;
282 g_user_access_recover_ip = reinterpret_cast<uint64_t>(&&recover_str);
283 NEXIOS_FAULT_RECOVERY_KEEP(recover_str);
284 arch::stac();
285 for (uint64_t i = 0; i < max_len; ++i) {
286 dst[i] = src[i];
287 if (src[i] == '\0') {
288 arch::clac();
289 g_user_access_recover_ip = 0;
290 return true;
291 }
292 }
293 dst[max_len - 1] = '\0';
294 arch::clac();
295 g_user_access_recover_ip = 0;
296 return true;
297
298recover_str:
299 arch::clac();
300 g_user_access_recover_ip = 0;
301 if (max_len > 0)
302 dst[0] = '\0';
303 return false;
304}
