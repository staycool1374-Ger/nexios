241 // Issue #271 P1: retained golden set (at most one). Adopted at
242 // take_completed from the taken TCB's live mappings (never during
243 // COPYING); abandoned on take-overwrite/reset/release. Golden phys
244 // are owned by mapping tasks, never by retention: the list + refs
245 // decide frees (zero/zero frees inline in the ref-drop). The pin
246 // (bounded RW file-byte copy) is retention-owned static storage.
247 struct GoldenPage {
248 uint64_t phys = 0;
249 bool executable = false;
250 };
251 struct GoldenRange {
252 uint64_t vaddr = 0;
253 uint64_t page_count = 0;
254 };
255 struct PinRange {
256 uint64_t vaddr = 0; // destination VA in the fresh build
257 uint64_t pin_off = 0; // offset into pin_buf_
258 uint64_t len = 0; // file bytes (never BSS tail)
259 uint64_t memsz = 0; // full RW span incl. BSS tail (fresh-mapped + zeroed)
260 };
261 struct Retention {
262 bool valid = false;
263 char path[kMaxPath] = {};
264 uint64_t file_size = 0;
265 uint64_t entry = 0;
266 // Packed header image for rerun finalization: real header with
267 // phoff rewritten to sizeof(ELF64Header) + packed table, so
268 // install_segment_canaries (hdr->phoff based) reads correctly.
269 ELF64Header hdr = {};
270 uint8_t phdr_image[sizeof(ELF64Header) +
271 64 * sizeof(ELF64ProgramHeader)] = {};
272 bool has_dynamic = false;
273 GoldenPage golden[kMaxGoldenPages] = {};
274 size_t golden_pages = 0;
275 GoldenRange ranges[kMaxGoldenRanges] = {};
276 size_t range_count = 0;
277 uint8_t pin_buf[kMaxRerunWritableBytes] = {};
278 size_t pin_len = 0;
279 PinRange pin_ranges[kMaxPinRanges] = {};
280 size_t pin_count = 0;
281 uint64_t task_refs = 0; // live tasks mapping the set
282 uint64_t epoch = 0; // bumped on every adopt/abandon (builder race detector)
283 bool retained = false; // retention holds a ref
284 RunOutcome outcome = RunOutcome::UNKNOWN;
285 uint64_t outcome_code = 0;
286 uint64_t outcome_task = 0;
287 uint32_t ro_crc32 = 0; // retained #46 original (compare, never adopt)
288 TaskControlBlock::RoRange ro_ranges[TaskControlBlock::kMaxRoRanges] = {};
289 size_t ro_count = 0;
290 uint64_t canary_slot = 0; // TEXT before-canary VA (P2 re-scan carve)
291 };
292 static Retention retention_;
293 static Retention staging_;
294 // Issue #271 P1: retention internals. All assume lock_ held +
295 // IRQ-masked by the caller; never reschedule inside (bounded walks
296 // and struct copies only; PMM frees mirror the destroy-under-lock
297 // precedent in reset()).
298 static void abandon_retention_locked();

302 // Drop the try_rerun builder ref ONLY if retention still refers to
303 // the epoch the builder snapshotted: a concurrent adopt swaps the
304 // whole Retention (new epoch, new owner), and an unconditional drop
305 // would corrupt the new set's count. Assumes lock_ held + IRQ-masked.
306 static void drop_builder_ref_locked(uint64_t epoch);
307 // Unwind a half-built rerun table: unmap retained-golden VAs (no ref

329 static uint32_t load_crc_;
