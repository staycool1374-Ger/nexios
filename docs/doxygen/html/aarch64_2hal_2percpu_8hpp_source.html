44struct alignas(arch::PAGE_SIZE) PerCpu {
45 // ─── ABI-frozen slots (mirrors x86 gs:0x00..0x08) ───────────────────
46 uint64_t user_rsp; // 0x00 — saved user RSP (syscall entry)
47 uint64_t kernel_rsp; // 0x08 — loaded kernel stack (ISR epilogue)
48
49 // ─── CPU identity ──────────────────────────────────────────────────
50 uint64_t cpu_id; // 0x10 — logical CPU index (0..CONFIG_MAX_CPUS-1)
51 uint64_t mpidr; // 0x18 — MPIDR_EL1 value (affinity levels)
52
53 // ─── Migrated from globals (matches x86 offsets) ───────────────────
54 uint64_t isr_nesting_depth; // 0x20 — was global isr_nesting_depth
55 uint64_t irq_entry_tsc; // 0x28 — was global irq_entry_tsc
56
57 // ─── FPU / scheduler ───────────────────────────────────────────────
58 // Own-CPU FPU register owner (issue #151): which task's FPU state is
59 // currently live in this CPU's registers. nullptr = none since boot.
60 void *fpu_owner; // 0x30 — per-CPU (was global)
61 void *current_task; // 0x38 — RESERVED (superseded by CpuContext
62 // array, issue #25 C1; kept for layout)
63
64 // ─── Interrupt prioritization (reserved on AArch64) ────────────────
65 // x86 carries the TPR shadow here (gs:0x40); AArch64 has no TPR, so
66 // this slot is reserved to keep the cross-arch layout identical.
67 uint64_t tpr_shadow; // 0x40 — reserved (x86: TPR class shadow)
68
69 // ─── Reserved / future expansion ───────────────────────────────────
70 // NOTE: explicit fields total 483 u64s; alignas(PAGE_SIZE) tail-pads
71 // sizeof to 4096 (enforced by the static_assert below). Keep the
72 // explicit total constant when adding fields (shrink reserved).
73 uint64_t reserved[474];
74};

77 "PerCpu must be exactly one 4 KiB page");

117inline PerCpu *per_cpu_current() {
118 const uint64_t base = read_tpidr_el1();
119 const uint64_t array_lo = reinterpret_cast<uint64_t>(&per_cpu[0]);
120 if (base >= array_lo && base < array_lo + sizeof(per_cpu)) {
121 return &per_cpu[(base - array_lo) / arch::PAGE_SIZE];
122 }
123 return &per_cpu[0];
124}
