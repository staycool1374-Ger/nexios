42inline uint64_t arch_page_table_current() {
43 uint64_t satp{};
44 asm volatile("csrr %0, satp" : "=r"(satp) : : "memory");
45 return (satp & 0xFFFFFFFFFFF) << 12; // PPN -> physical address
46}

50inline void arch_page_table_activate(uint64_t phys_ppn_shifted) {
51 // Input is physical address >> 12 (PPN)
52 uint64_t satp = (8ULL << 60) | (phys_ppn_shifted >> 12); // Sv39 mode | PPN
53 asm volatile("csrw satp, %0" : : "r"(satp) : "memory");
54 asm volatile("sfence.vma" : : : "memory");

175inline uint64_t *ArchPageTable::get_table(uint64_t table_base, uint64_t index,
176 bool create) {
177 uint64_t *table = reinterpret_cast<uint64_t *>(table_base);
178 uint64_t entry = table[index];
179
180 if (entry & DESC_VALID) {
181 return reinterpret_cast<uint64_t *>(entry & ~0xFFF);
182 }
183
184 if (!create)
185 return nullptr;
186
187 uint64_t new_page = kernel::PMM::alloc_page_table();
188 if (!new_page)
189 return nullptr;
190
191 void *new_page_ptr = reinterpret_cast<void *>(new_page);
192 memset(new_page_ptr, 0, PAGE_SIZE);
193 // Mark as valid, readable, writable table pointer
194 table[index] =
195 (new_page & ~0xFFF) | DESC_VALID | DESC_TABLE | DESC_R | DESC_W;
196 return reinterpret_cast<uint64_t *>(new_page);
197}

202inline uint64_t ArchPageTable::attr_from_flags(uint64_t flags) {
203 uint64_t attr = DESC_VALID | DESC_AF;
204 (void)DESC_G;
205 if (flags & kernel::PageFlags::PRESENT)
206 attr |= DESC_VALID;
207 if (!(flags & kernel::PageFlags::NX))
208 attr |= DESC_X;
209 if (flags & kernel::PageFlags::WRITE)
210 attr |= DESC_W;
211 if (flags & kernel::PageFlags::USER)
212 attr |= DESC_USER;
213 // Default to readable unless explicitly no-access
214 if (flags & kernel::PageFlags::PRESENT)
215 attr |= DESC_R;
216 return attr;
217}

254inline kernel::ErrorOr<void>
255ArchPageTable::map_page(uint64_t virt, uint64_t phys, uint64_t flags) {
256 uint64_t l0_idx = (virt >> L0_SHIFT) & TABLE_MASK;
257 uint64_t l1_idx = (virt >> L1_SHIFT) & TABLE_MASK;
258 uint64_t l2_idx = (virt >> L2_SHIFT) & TABLE_MASK;
259
260 uint64_t l0_phys = current();
261 uint64_t *l1 = walk_l0(l0_phys, l0_idx, true);
262 if (!l1)
263 return kernel::Error::OOM;
264
265 uint64_t *l2 = walk_l1(reinterpret_cast<uint64_t>(l1), l1_idx, true);
266 if (!l2)
267 return kernel::Error::OOM;
268
269 uint64_t entry_flags = attr_from_flags(flags);
270
271 // Try 2MB block mapping if aligned
272 if ((virt & ((1ULL << L1_SHIFT) - 1)) == 0 &&
273 (phys & ((1ULL << L1_SHIFT) - 1)) == 0) {
274 l2[l1_idx] =
275 (phys & ~0x1FFFFF) | entry_flags | DESC_BLOCK | (1ULL << 6);
276 return {};
277 }
278
279 // 4KB page at L2
280 l2[l2_idx] =
281 (phys & ~0xFFF) | entry_flags | DESC_PAGE | (1ULL << 6) | (1ULL << 7);
282 return {};
283}

288inline kernel::ErrorOr<void> ArchPageTable::unmap_page(uint64_t virt) {
289 uint64_t l0_idx = (virt >> L0_SHIFT) & TABLE_MASK;
290 uint64_t l1_idx = (virt >> L1_SHIFT) & TABLE_MASK;
291 uint64_t l2_idx = (virt >> L2_SHIFT) & TABLE_MASK;
292
293 uint64_t l0_phys = current();
294 uint64_t *l0 = reinterpret_cast<uint64_t *>(l0_phys);
295
296 if (!(l0[l0_idx] & DESC_VALID))
297 return kernel::Error::NOT_FOUND;
298 uint64_t *l1 = reinterpret_cast<uint64_t *>(l0[l0_idx] & ~0xFFF);
299 if (!(l1[l1_idx] & DESC_VALID))
300 return kernel::Error::NOT_FOUND;
301 uint64_t *l2 = reinterpret_cast<uint64_t *>(l1[l1_idx] & ~0xFFF);
302
303 // Check for 2MB block
304 if ((virt & ((1ULL << L1_SHIFT) - 1)) == 0 && !(l2[l1_idx] & DESC_TABLE)) {
305 l2[l1_idx] = 0;
306 return {};
307 }
308
309 if (!(l2[l2_idx] & DESC_VALID))
310 return kernel::Error::NOT_FOUND;
311 l2[l2_idx] = 0;
312 return {};
313}

319inline uint64_t ArchPageTable::get_physical(uint64_t virt) {
320 uint64_t l0_idx = (virt >> L0_SHIFT) & TABLE_MASK;
321 uint64_t l1_idx = (virt >> L1_SHIFT) & TABLE_MASK;
322 uint64_t l2_idx = (virt >> L2_SHIFT) & TABLE_MASK;
323
324 uint64_t l0_phys = current();
325 uint64_t *l0 = reinterpret_cast<uint64_t *>(l0_phys);
326
327 if (!(l0[l0_idx] & DESC_VALID))
328 return 0;
329 uint64_t *l1 = reinterpret_cast<uint64_t *>(l0[l0_idx] & ~0xFFF);
330 if (!(l1[l1_idx] & DESC_VALID))
331 return 0;
332 uint64_t *l2 = reinterpret_cast<uint64_t *>(l1[l1_idx] & ~0xFFF);
333
334 // 2MB block mapping
335 if ((virt & ((1ULL << L1_SHIFT) - 1)) == 0 && !(l2[l1_idx] & DESC_TABLE)) {
336 return (l2[l1_idx] & ~0x1FFFFF) | (virt & 0x1FFFFF);
337 }
338
339 if (!(l2[l2_idx] & DESC_VALID))
340 return 0;
341 return (l2[l2_idx] & ~0xFFF) | (virt & 0xFFF);
342}
